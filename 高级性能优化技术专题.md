## 1. 现代硬件加速器

### 1.1 Intel DSA (Data Streaming Accelerator)

Intel DSA是Sapphire Rapids及更新处理器中的硬件加速器，用于加速数据移动和转换操作。

#### DSA架构概览

```
CPU Cores ──┐
            ├─→ DSA Engine 1 ──→ Memory Copy/Fill
            ├─→ DSA Engine 2 ──→ CRC/DIF
            ├─→ DSA Engine 3 ──→ Compare
            └─→ DSA Engine 4 ──→ Delta Record
                    ↓
              Shared Work Queue
                    ↓
              Memory Subsystem
```

#### 使用DSA加速内存拷贝

```c
// dsa_memcpy.c
// 使用Intel DSA加速大块内存拷贝
#include <linux/idxd.h>
#include <sys/mman.h>
#include <fcntl.h>

struct dsa_context {
    int fd;
    void* portal;
    struct dsa_hw_desc* desc;
};

int dsa_init(struct dsa_context* ctx) {
    // 打开DSA设备
    ctx->fd = open("/dev/dsa/wq0.0", O_RDWR);
    if (ctx->fd < 0) {
        return -1;
    }

    // 映射提交门户
    ctx->portal = mmap(NULL, 0x1000, PROT_WRITE,
                       MAP_SHARED | MAP_POPULATE, ctx->fd, 0);
    if (ctx->portal == MAP_FAILED) {
        close(ctx->fd);
        return -1;
    }

    return 0;
}

// 异步内存拷贝
int dsa_memcpy_async(struct dsa_context* ctx,
                     void* dst, const void* src, size_t size) {
    struct dsa_hw_desc desc = {0};

    // 设置操作码：内存拷贝
    desc.opcode = DSA_OPCODE_MEMMOVE;
    desc.flags = IDXD_OP_FLAG_CRAV | IDXD_OP_FLAG_RCR;

    // 设置源和目标地址
    desc.src_addr = (uint64_t)src;
    desc.dst_addr = (uint64_t)dst;
    desc.xfer_size = size;

    // 提交到DSA
    __builtin_ia32_movdir64b(ctx->portal, &desc);

    return 0;
}

// 批量操作示例
void dsa_batch_memcpy(struct dsa_context* ctx,
                      void** dsts, void** srcs,
                      size_t* sizes, int count) {
    // 提交多个描述符
    for (int i = 0; i < count; i++) {
        dsa_memcpy_async(ctx, dsts[i], srcs[i], sizes[i]);
    }

    // DSA会并行处理这些操作
}

// 性能对比
void benchmark_dsa_vs_memcpy() {
    const size_t SIZE = 1024 * 1024 * 1024; // 1GB
    void* src = aligned_alloc(64, SIZE);
    void* dst1 = aligned_alloc(64, SIZE);
    void* dst2 = aligned_alloc(64, SIZE);

    struct dsa_context dsa;
    dsa_init(&dsa);

    // 标准memcpy: ~12 GB/s (单核)
    clock_t start = clock();
    memcpy(dst1, src, SIZE);
    double time1 = (double)(clock() - start) / CLOCKS_PER_SEC;

    // DSA: ~30-50 GB/s (硬件加速)
    start = clock();
    dsa_memcpy_async(&dsa, dst2, src, SIZE);
    // 等待完成
    dsa_wait_completion(&dsa);
    double time2 = (double)(clock() - start) / CLOCKS_PER_SEC;

    printf("memcpy: %.2f GB/s\n", SIZE / time1 / 1e9);
    printf("DSA:    %.2f GB/s\n", SIZE / time2 / 1e9);
}
```

```c
// 1. 使用批处理提高吞吐量
void efficient_dsa_copy(struct dsa_context* ctx,
                       struct copy_task* tasks, int count) {
    const int BATCH_SIZE = 16;

    for (int i = 0; i < count; i += BATCH_SIZE) {
        int batch = (count - i < BATCH_SIZE) ? (count - i) : BATCH_SIZE;

        // 批量提交
        for (int j = 0; j < batch; j++) {
            dsa_memcpy_async(ctx,
                           tasks[i+j].dst,
                           tasks[i+j].src,
                           tasks[i+j].size);
        }

        // 批量等待完成
        dsa_wait_batch(ctx, batch);
    }
}

// 2. 对齐优化
void* dsa_aligned_alloc(size_t size) {
    // DSA对64字节对齐的数据性能最佳
    return aligned_alloc(64, (size + 63) & ~63);
}

// 3. 避免小块传输
bool should_use_dsa(size_t size) {
    // DSA对于小于4KB的传输开销较大
    // 最佳使用场景：>= 64KB
    return size >= 65536;
}
```

#### QAT压缩加速

```c
// qat_compression.c
// 使用Intel QAT加速数据压缩
#include <qat/cpa.h>
#include <qat/cpa_dc.h>

typedef struct {
    CpaInstanceHandle instance;
    CpaDcSessionHandle session;
} QATContext;

// 初始化QAT压缩
int qat_compress_init(QATContext* ctx) {
    CpaStatus status;
    Cpa16U num_instances = 0;

    // 获取压缩实例数量
    status = cpaDcGetNumInstances(&num_instances);
    if (status != CPA_STATUS_SUCCESS || num_instances == 0) {
        return -1;
    }

    // 获取实例句柄
    CpaInstanceHandle* instances = malloc(num_instances * sizeof(CpaInstanceHandle));
    cpaDcGetInstances(num_instances, instances);
    ctx->instance = instances[0];
    free(instances);

    // 启动实例
    cpaDcStartInstance(ctx->instance);

    // 创建会话
    CpaDcSessionSetupData setup = {0};
    setup.compLevel = CPA_DC_L4;  // 压缩级别
    setup.compType = CPA_DC_DEFLATE;
    setup.huffType = CPA_DC_HT_FULL_DYNAMIC;
    setup.sessDirection = CPA_DC_DIR_COMPRESS;

    Cpa32U session_size = 0;
    cpaDcGetSessionSize(ctx->instance, &setup, &session_size);

    ctx->session = malloc(session_size);
    status = cpaDcInitSession(ctx->instance, ctx->session, &setup, NULL, NULL);

    return (status == CPA_STATUS_SUCCESS) ? 0 : -1;
}

// 异步压缩回调
void qat_compress_callback(void* callback_tag,
                          CpaStatus status,
                          CpaDcRqResults* results) {
    struct compress_request* req = (struct compress_request*)callback_tag;

    if (status == CPA_STATUS_SUCCESS) {
        req->compressed_size = results->produced;
        req->consumed_size = results->consumed;
    }

    req->completed = 1;
    sem_post(&req->completion_sem);
}

// 压缩数据
int qat_compress(QATContext* ctx,
                const void* src, size_t src_len,
                void* dst, size_t dst_len,
                size_t* compressed_size) {
    CpaStatus status;

    // 分配缓冲区元数据
    CpaBufferList src_buffer = {0};
    CpaBufferList dst_buffer = {0};
    CpaFlatBuffer src_flat = {0};
    CpaFlatBuffer dst_flat = {0};

    src_flat.pData = (Cpa8U*)src;
    src_flat.dataLenInBytes = src_len;
    src_buffer.pBuffers = &src_flat;
    src_buffer.numBuffers = 1;

    dst_flat.pData = (Cpa8U*)dst;
    dst_flat.dataLenInBytes = dst_len;
    dst_buffer.pBuffers = &dst_flat;
    dst_buffer.numBuffers = 1;

    // 执行压缩
    CpaDcRqResults results = {0};
    struct compress_request req = {0};
    sem_init(&req.completion_sem, 0, 0);

    status = cpaDcCompressData2(
        ctx->instance,
        ctx->session,
        &src_buffer,
        &dst_buffer,
        &results,
        CPA_DC_FLUSH_FINAL,
        &req  // 回调上下文
    );

    if (status == CPA_STATUS_SUCCESS) {
        // 等待完成
        sem_wait(&req.completion_sem);
        *compressed_size = req.compressed_size;
        return 0;
    }

    return -1;
}

// 性能对比：QAT vs zlib
void benchmark_qat_vs_zlib() {
    const size_t SIZE = 100 * 1024 * 1024; // 100MB

    double time1 = (double)(clock() - start) / CLOCKS_PER_SEC;

    // QAT压缩: ~2-5 GB/s (硬件加速)
    QATContext qat;
    qat_compress_init(&qat);

    start = clock();
    size_t compressed2 = 0;
    qat_compress(&qat, src, SIZE, dst2, SIZE, &compressed2);
    double time2 = (double)(clock() - start) / CLOCKS_PER_SEC;

    printf("zlib: %.2f MB/s, ratio: %.2f%%\n",
           SIZE / time1 / 1e6, 100.0 * compressed1 / SIZE);
```

### 1.3 RDMA (Remote Direct Memory Access)

RDMA允许网络适配器直接读写远程内存，绕过CPU和操作系统内核。

#### RDMA基础操作

```c
// rdma_example.c
// RDMA高性能网络通信
#include <rdma/rdma_cma.h>
#include <infiniband/verbs.h>

struct rdma_context {
    struct rdma_cm_id* cm_id;
    struct ibv_pd* pd;
    struct ibv_cq* cq;
    struct ibv_qp* qp;
    struct ibv_mr* mr;
    void* buffer;
    size_t buffer_size;
};

// 初始化RDMA上下文
int rdma_init_client(struct rdma_context* ctx,
                     const char* server_addr, int port) {
    struct rdma_addrinfo hints = {0}, *res;
    char port_str[16];
    int ret;

    // 解析地址
    hints.ai_port_space = RDMA_PS_TCP;
    snprintf(port_str, sizeof(port_str), "%d", port);

    ret = rdma_getaddrinfo(server_addr, port_str, &hints, &res);
    if (ret) {
        return -1;
    }

    // 创建CM ID
    ret = rdma_create_ep(&ctx->cm_id, res, NULL, NULL);
    rdma_freeaddrinfo(res);
    if (ret) {
        return -1;
    }

    // 连接到服务器
    ret = rdma_connect(ctx->cm_id, NULL);
    if (ret) {
        rdma_destroy_ep(ctx->cm_id);
        return -1;
    }

    // 获取设备上下文
    struct ibv_context* verbs = ctx->cm_id->verbs;

    // 分配保护域
    ctx->pd = ibv_alloc_pd(verbs);

    // 创建完成队列
    ctx->cq = ibv_create_cq(verbs, 128, NULL, NULL, 0);

    // 注册内存区域
    ctx->buffer_size = 1024 * 1024; // 1MB
    ctx->buffer = aligned_alloc(4096, ctx->buffer_size);

    ctx->mr = ibv_reg_mr(ctx->pd, ctx->buffer, ctx->buffer_size,
                        IBV_ACCESS_LOCAL_WRITE |
                        IBV_ACCESS_REMOTE_READ |
                        IBV_ACCESS_REMOTE_WRITE);

    return 0;
}

// RDMA WRITE - 单边操作
int rdma_write(struct rdma_context* ctx,
              void* local_addr, size_t length,
              uint64_t remote_addr, uint32_t remote_rkey) {
    struct ibv_send_wr wr = {0};
    struct ibv_send_wr* bad_wr = NULL;
    struct ibv_sge sge;

    // 设置scatter-gather entry
    sge.addr = (uint64_t)local_addr;
    sge.length = length;
    sge.lkey = ctx->mr->lkey;

    // 设置work request
    wr.wr_id = 0;
    wr.sg_list = &sge;
    wr.num_sge = 1;
    wr.opcode = IBV_WR_RDMA_WRITE;
    wr.send_flags = IBV_SEND_SIGNALED;
    wr.wr.rdma.remote_addr = remote_addr;
    wr.wr.rdma.rkey = remote_rkey;

    // 提交work request
    return ibv_post_send(ctx->cm_id->qp, &wr, &bad_wr);
}

// RDMA READ - 单边操作
int rdma_read(struct rdma_context* ctx,
             void* local_addr, size_t length,
             uint64_t remote_addr, uint32_t remote_rkey) {
    struct ibv_send_wr wr = {0};
    struct ibv_send_wr* bad_wr = NULL;
    struct ibv_sge sge;

    sge.addr = (uint64_t)local_addr;
    sge.length = length;
    sge.lkey = ctx->mr->lkey;

    wr.wr_id = 0;
    wr.sg_list = &sge;
    wr.num_sge = 1;
    wr.opcode = IBV_WR_RDMA_READ;
    wr.send_flags = IBV_SEND_SIGNALED;
    wr.wr.rdma.remote_addr = remote_addr;
    wr.wr.rdma.rkey = remote_rkey;

    return ibv_post_send(ctx->cm_id->qp, &wr, &bad_wr);
}

// 轮询完成
int rdma_poll_completion(struct rdma_context* ctx) {
    struct ibv_wc wc;
    int ret;

    do {
        ret = ibv_poll_cq(ctx->cq, 1, &wc);
        if (ret > 0) {
            if (wc.status == IBV_WC_SUCCESS) {
                return 0;
            } else {
                fprintf(stderr, "RDMA operation failed: %s\n",
                       ibv_wc_status_str(wc.status));
                return -1;
            }
        }
    } while (ret == 0);

    return -1;
}

// RDMA性能测试
void benchmark_rdma_vs_tcp() {
    const size_t SIZE = 1024 * 1024; // 1MB
    const int ITERATIONS = 10000;

    struct rdma_context rdma;
    rdma_init_client(&rdma, "192.168.1.100", 12345);

    // TCP传输: ~10 Gbps, ~5 μs latency
    // RDMA传输: ~100 Gbps, ~1 μs latency

    clock_t start = clock();
    for (int i = 0; i < ITERATIONS; i++) {
        rdma_write(&rdma, rdma.buffer, SIZE,
                  remote_addr, remote_rkey);
        rdma_poll_completion(&rdma);
    }
    double time = (double)(clock() - start) / CLOCKS_PER_SEC;

    double throughput = (SIZE * ITERATIONS) / time / 1e9;
    double latency = time * 1e6 / ITERATIONS;

    printf("RDMA Throughput: %.2f GB/s\n", throughput);
    printf("RDMA Latency: %.2f μs\n", latency);
}
```

```c
// 1. 使用Inline Data减少小消息延迟
int rdma_send_inline(struct rdma_context* ctx,
                     void* data, size_t length) {
    // 对于小数据（<= 256字节），使用inline可以避免DMA
    if (length <= 256) {
        struct ibv_send_wr wr = {0};
        struct ibv_sge sge;

        sge.addr = (uint64_t)data;
        sge.length = length;
        sge.lkey = ctx->mr->lkey;

        wr.sg_list = &sge;
        wr.num_sge = 1;
        wr.opcode = IBV_WR_SEND;
        wr.send_flags = IBV_SEND_INLINE | IBV_SEND_SIGNALED;

        struct ibv_send_wr* bad_wr;
        return ibv_post_send(ctx->qp, &wr, &bad_wr);
    }

    // 大数据使用正常发送
    return rdma_send_normal(ctx, data, length);
}

// 2. 批量操作提高吞吐量
int rdma_write_batch(struct rdma_context* ctx,
                    struct iovec* iov, int count,
                    uint64_t remote_addr, uint32_t remote_rkey) {
    struct ibv_send_wr wrs[count];
    struct ibv_send_wr* bad_wr;
    struct ibv_sge sges[count];

    // 构建work request链
    for (int i = 0; i < count; i++) {
        sges[i].addr = (uint64_t)iov[i].iov_base;
        sges[i].length = iov[i].iov_len;
        sges[i].lkey = ctx->mr->lkey;

        wrs[i].wr_id = i;
        wrs[i].sg_list = &sges[i];
        wrs[i].num_sge = 1;
        wrs[i].opcode = IBV_WR_RDMA_WRITE;
        wrs[i].send_flags = (i == count - 1) ? IBV_SEND_SIGNALED : 0;
        wrs[i].wr.rdma.remote_addr = remote_addr;
        wrs[i].wr.rdma.rkey = remote_rkey;
        wrs[i].next = (i < count - 1) ? &wrs[i + 1] : NULL;

        remote_addr += iov[i].iov_len;
    }

    // 一次性提交所有请求
    return ibv_post_send(ctx->qp, &wrs[0], &bad_wr);
}

// 3. 使用Unsignaled模式减少CQ轮询
#define UNSIGNAL_BATCH 64

int rdma_write_unsignaled(struct rdma_context* ctx,
                         void* data, size_t length,
                         uint64_t remote_addr, uint32_t remote_rkey,
                         int* op_count) {
    struct ibv_send_wr wr = {0};
    struct ibv_send_wr* bad_wr;
    struct ibv_sge sge;

    sge.addr = (uint64_t)data;
    sge.length = length;
    sge.lkey = ctx->mr->lkey;

    wr.sg_list = &sge;
    wr.num_sge = 1;
    wr.opcode = IBV_WR_RDMA_WRITE;
    wr.wr.rdma.remote_addr = remote_addr;
    wr.wr.rdma.rkey = remote_rkey;

    // 每UNSIGNAL_BATCH个操作才signal一次
    (*op_count)++;
    if (*op_count % UNSIGNAL_BATCH == 0) {
        wr.send_flags = IBV_SEND_SIGNALED;
        *op_count = 0;
    }

    return ibv_post_send(ctx->qp, &wr, &bad_wr);
}
```

```c
struct persistent_data {

    char data[1024];

int optane_direct_access() {
    size_t mapped_len;
    int is_pmem;

    // 映射持久内存
    struct persistent_data* pdata = pmem_map_file(
        "/mnt/pmem0/mydata",
        sizeof(struct persistent_data),
        PMEM_FILE_CREATE,
        0666,
        &mapped_len,
        &is_pmem
    );

    if (!pdata) {
        return -1;
    }

    // 原子更新（确保崩溃一致性）
    pdata->counter++;

    // 刷新到持久内存
    if (is_pmem) {
        pmem_persist(&pdata->counter, sizeof(pdata->counter));
    } else {
        pmem_msync(&pdata->counter, sizeof(pdata->counter));

    pmem_unmap(pdata, mapped_len);
    return 0;

            _mm_clwb((void*)p);  // 写回缓存行
        }
        _mm_sfence();
    }
```

### 2.1 中断亲和性优化

```bash
#!/bin/bash
# irq_affinity.sh
# 优化网卡中断分布，避免跨NUMA节点

# 获取网卡的IRQ列表
get_irqs() {
    local DEVICE=$1
    grep $DEVICE /proc/interrupts | awk '{print $1}' | sed 's/://g'
}

# 设置IRQ亲和性到特定CPU
set_irq_affinity() {
    local IRQ=$1
    local CPU=$2

    # 计算CPU掩码
    printf "%x" $((1 << $CPU)) > /proc/irq/$IRQ/smp_affinity

    # 禁用irqbalance对此IRQ的干预
    echo 0 > /proc/irq/$IRQ/smp_affinity_hint
}

# 智能分配：将网卡队列绑定到同NUMA节点的CPU
optimize_network_irqs() {
    local DEVICE="eth0"
    local NUMA_NODE=0

    # 获取该NUMA节点的CPU列表
    CPUS=$(cat /sys/devices/system/node/node${NUMA_NODE}/cpulist)

    # 获取网卡IRQ
    IRQS=$(get_irqs $DEVICE)

    # 将IRQ均匀分布到CPU
    IRQ_ARRAY=($IRQS)
    CPU_ARRAY=($(seq $(echo $CPUS | tr ',' ' ')))

    for i in "${!IRQ_ARRAY[@]}"; do
        IRQ=${IRQ_ARRAY[$i]}
        CPU=${CPU_ARRAY[$((i % ${#CPU_ARRAY[@]}))]}
        set_irq_affinity $IRQ $CPU
        echo "IRQ $IRQ -> CPU $CPU"
    done
}

# 禁用CPU节能（避免C-state影响延迟）
disable_cpu_idle() {
    for i in /sys/devices/system/cpu/cpu*/cpuidle/state*/disable; do
        echo 1 > $i
    done
}

# RPS/RFS配置（接收包分发）
configure_rps_rfs() {
    local DEVICE="eth0"

    # 启用RPS - 将接收处理分散到多个CPU
    for queue in /sys/class/net/$DEVICE/queues/rx-*/rps_cpus; do
        echo "ff" > $queue  # 使用所有CPU
    done

    # 启用RFS - 将数据包路由到处理该连接的CPU
    echo 32768 > /proc/sys/net/core/rps_sock_flow_entries
    for queue in /sys/class/net/$DEVICE/queues/rx-*/rps_flow_cnt; do
        echo 2048 > $queue
    done

configure_xps() {
    local DEVICE="eth0"
    local NUM_QUEUES=$(ls -d /sys/class/net/$DEVICE/queues/tx-* | wc -l)

    # 将每个TX队列绑定到对应CPU

        printf "%x" $((1 << $i)) > /sys/class/net/$DEVICE/queues/tx-$i/xps_cpus
    done
}
```

```c
// 利用CPU的乱序执行和指令级并行

// ❌ 差：数据依赖链
void sum_dependent(const int* a, int n) {
    int sum = 0;
    for (int i = 0; i < n; i++) {
        sum += a[i];  // 每次加法依赖前一次结果
    }
    // ILP = 1，每个周期只能执行1条加法
}

// ✅ 好：打破依赖链
void sum_independent(const int* a, int n) {
    int sum0 = 0, sum1 = 0, sum2 = 0, sum3 = 0;

    for (int i = 0; i < n; i += 4) {
        sum0 += a[i];
        sum1 += a[i+1];
        sum2 += a[i+2];
        sum3 += a[i+3];
    }

    return sum0 + sum1 + sum2 + sum3;
    // ILP = 4，4条加法可以并行执行
    // 性能提升：2-3x
}

// 更复杂的例子：矩阵乘法
// ❌ 差：单累加器
void matmul_slow(float* C, const float* A, const float* B, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            float sum = 0;
            for (int k = 0; k < n; k++) {
                sum += A[i*n + k] * B[k*n + j];
            }
            C[i*n + j] = sum;
        }
    }
}

// ✅ 好：多累加器 + 循环展开
void matmul_fast(float* C, const float* A, const float* B, int n) {
    for (int i = 0; i < n; i++) {
        for (int j = 0; j < n; j++) {
            float sum0 = 0, sum1 = 0, sum2 = 0, sum3 = 0;

            int k;
            for (k = 0; k < n - 3; k += 4) {
                sum0 += A[i*n + k] * B[k*n + j];
                sum1 += A[i*n + k+1] * B[(k+1)*n + j];
                sum2 += A[i*n + k+2] * B[(k+2)*n + j];
                sum3 += A[i*n + k+3] * B[(k+3)*n + j];
            }

            // 处理剩余
            for (; k < n; k++) {
                sum0 += A[i*n + k] * B[k*n + j];
            }

            C[i*n + j] = sum0 + sum1 + sum2 + sum3;
        }
    }
    // 性能提升：3-5x
}
```

```c
#include <bpf/bpf_helpers.h>

// BPF map：记录函数调用延迟
struct {
    __uint(type, BPF_MAP_TYPE_HASH);
    __type(key, u32);  // PID
    __type(value, u64);  // 开始时间
    __uint(max_entries, 10000);
} start_times SEC(".maps");

struct {
    __uint(type, BPF_MAP_TYPE_HISTOGRAM);
    __type(key, u64);  // 延迟（纳秒）
    __type(value, u64);  // 计数
    __uint(max_entries, 100);
} latency_hist SEC(".maps");

// 追踪函数入口
SEC("kprobe/do_sys_open")
int trace_open_entry(struct pt_regs* ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64 ts = bpf_ktime_get_ns();

    bpf_map_update_elem(&start_times, &pid, &ts, BPF_ANY);
    return 0;
}

// 追踪函数出口
SEC("kretprobe/do_sys_open")
int trace_open_return(struct pt_regs* ctx) {
    u32 pid = bpf_get_current_pid_tgid() >> 32;
    u64* start_ts = bpf_map_lookup_elem(&start_times, &pid);

    if (start_ts) {
        u64 end_ts = bpf_ktime_get_ns();
        u64 latency = end_ts - *start_ts;

        // 更新直方图
        u64 slot = latency / 1000;  // 微秒
        u64* count = bpf_map_lookup_elem(&latency_hist, &slot);
        if (count) {
            __sync_fetch_and_add(count, 1);
        } else {
            u64 one = 1;
            bpf_map_update_elem(&latency_hist, &slot, &one, BPF_ANY);
        }

        bpf_map_delete_elem(&start_times, &pid);
    }

    return 0;
}
```

```c
#include <sys/sendfile.h>

#include <sys/mman.h>

// 1. sendfile() - 内核空间零拷贝
ssize_t sendfile_example(int socket_fd, int file_fd, size_t count) {
    // 直接在内核中传输文件内容到socket
    // 避免用户空间拷贝
    off_t offset = 0;
    return sendfile(socket_fd, file_fd, &offset, count);

    // 传统方式需要4次拷贝 + 2次系统调用:
    // 1. DMA: 磁盘 -> 内核缓冲区
    // 2. CPU: 内核缓冲区 -> 用户缓冲区 (read)
    // 3. CPU: 用户缓冲区 -> socket缓冲区 (write)
    // 4. DMA: socket缓冲区 -> 网卡

    // sendfile只需2次拷贝 + 1次系统调用:
    // 1. DMA: 磁盘 -> 内核缓冲区
    // 2. DMA: 内核缓冲区 -> 网卡
    //性能提升：2-3x
}

// 2. splice() - 管道零拷贝
ssize_t splice_example(int fd_in, int fd_out, size_t len) {
    int pipefd[2];
    pipe(pipefd);

    // 从输入fd拷贝到管道
    ssize_t bytes = splice(fd_in, NULL, pipefd[1], NULL,
                           len, SPLICE_F_MOVE);

    // 从管道拷贝到输出fd
    splice(pipefd[0], NULL, fd_out, NULL, bytes, SPLICE_F_MOVE);

    close(pipefd[0]);
    close(pipefd[1]);

    return bytes;
}

// 3. mmap() + write() - 映射零拷贝
void mmap_send(int socket_fd, const char* filename) {
    int fd = open(filename, O_RDONLY);
    struct stat st;
    fstat(fd, &st);

    // 映射文件到内存
    void* mapped = mmap(NULL, st.st_size, PROT_READ,
                       MAP_PRIVATE, fd, 0);

    // 直接发送映射的内存
    send(socket_fd, mapped, st.st_size, 0);

    munmap(mapped, st.st_size);
    close(fd);
}

// 4. MSG_ZEROCOPY - socket零拷贝
int msg_zerocopy_send(int socket_fd, void* buffer, size_t len) {
    // 启用零拷贝发送
    int optval = 1;
    setsockopt(socket_fd, SOL_SOCKET, SO_ZEROCOPY,
              &optval, sizeof(optval));

    // 发送数据（异步）
    struct msghdr msg = {0};
    struct iovec iov = {
        .iov_base = buffer,
        .iov_len = len
    };
    msg.msg_iov = &iov;
    msg.msg_iovlen = 1;

    ssize_t sent = sendmsg(socket_fd, &msg, MSG_ZEROCOPY);

    // 等待发送完成通知
    struct sock_extended_err* serr;
    struct msghdr msg_err = {0};
    struct cmsghdr* cm;
    char control[100];

    msg_err.msg_control = control;
    msg_err.msg_controllen = sizeof(control);

    recvmsg(socket_fd, &msg_err, MSG_ERRQUEUE);

    // 检查完成状态
    for (cm = CMSG_FIRSTHDR(&msg_err); cm; cm = CMSG_NXTHDR(&msg_err, cm)) {
        if (cm->cmsg_level == SOL_IP && cm->cmsg_type == IP_RECVERR) {
            serr = (struct sock_extended_err*)CMSG_DATA(cm);
            if (serr->ee_errno == 0 && serr->ee_origin == SO_EE_ORIGIN_ZEROCOPY) {
                // 零拷贝发送成功
                return sent;
            }
        }
    }

    return -1;
}
```

```c
#define TX_RING_SIZE 1024

// 初始化DPDK端口

    struct rte_eth_conf port_conf = {0};
    const uint16_t rx_rings = 1, tx_rings = 1;
    uint16_t nb_rxd = RX_RING_SIZE;
    uint16_t nb_txd = TX_RING_SIZE;
    int retval;
    uint16_t q;
    struct rte_eth_dev_info dev_info;

    // 获取设备信息
    retval = rte_eth_dev_info_get(port, &dev_info);
    if (retval != 0) {

    }

    retval = rte_eth_dev_configure(port, rx_rings, tx_rings, &port_conf);
    if (retval != 0) {
        return retval;
    }

    // 配置RX队列
    for (q = 0; q < rx_rings; q++) {
        retval = rte_eth_rx_queue_setup(port, q, nb_rxd,
                rte_eth_dev_socket_id(port), NULL, mbuf_pool);
        if (retval < 0) {
            return retval;
        }
    }

    // 配置TX队列
    for (q = 0; q < tx_rings; q++) {

            return retval;
        }

    retval = rte_eth_dev_start(port);
    if (retval < 0) {
        return retval;
    }

    // 启用混杂模式
    rte_eth_promiscuous_enable(port);

}

// L2转发（最简单的例子）
void dpdk_l2_forward(void) {
    uint16_t port = 0;
    struct rte_mbuf* bufs[BURST_SIZE];
    uint16_t nb_rx, nb_tx;

    while (1) {
        // 批量接收包
        nb_rx = rte_eth_rx_burst(port, 0, bufs, BURST_SIZE);

        if (unlikely(nb_rx == 0))
            continue;

        // 处理包（例如：修改MAC地址）
        for (int i = 0; i < nb_rx; i++) {
            struct rte_ether_hdr* eth_hdr =
                rte_pktmbuf_mtod(bufs[i], struct rte_ether_hdr*);

            // 交换源和目标MAC
            struct rte_ether_addr tmp;
            rte_ether_addr_copy(&eth_hdr->src_addr, &tmp);
            rte_ether_addr_copy(&eth_hdr->dst_addr, &eth_hdr->src_addr);
            rte_ether_addr_copy(&tmp, &eth_hdr->dst_addr);
        }

        // 批量发送包
        nb_tx = rte_eth_tx_burst(port, 0, bufs, nb_rx);

        // 释放未发送的包
        if (unlikely(nb_tx < nb_rx)) {
            for (int i = nb_tx; i < nb_rx; i++) {
                rte_pktmbuf_free(bufs[i]);
            }
        }
    }
}
```

```c
// XDP在网卡驱动层处理包，比iptables快10-100倍

SEC("xdp")
int xdp_firewall(struct xdp_md* ctx) {
    void* data_end = (void*)(long)ctx->data_end;
    void* data = (void*)(long)ctx->data;

    // 解析以太网头
    struct ethhdr* eth = data;
    if ((void*)(eth + 1) > data_end) {
        return XDP_DROP;
    }

    // 只处理IPv4
    if (eth->h_proto != __constant_htons(ETH_P_IP)) {
        return XDP_PASS;
    }

    // 解析IP头
    struct iphdr* ip = (void*)(eth + 1);
    if ((void*)(ip + 1) > data_end) {
        return XDP_DROP;
    }

    // DDoS防护：限制来自单一源的包速率
    __u32 src_ip = ip->saddr;

    // 黑名单过滤
    if (is_blacklisted(src_ip)) {
        return XDP_DROP;  // 直接丢弃，无需到达内核
    }

    // TCP SYN flood防护
    if (ip->protocol == IPPROTO_TCP) {
        struct tcphdr* tcp = (void*)ip + sizeof(*ip);
        if ((void*)(tcp + 1) > data_end) {
            return XDP_DROP;
        }

        if (tcp->syn && !tcp->ack) {
            // SYN包速率限制
            if (rate_limit_syn(src_ip)) {
                return XDP_DROP;
            }
        }
    }

    // 转发到常规网络栈
    return XDP_PASS;
}

// XDP_TX：原路返回包（高效的echo server）
SEC("xdp")
int xdp_echo(struct xdp_md* ctx) {
    void* data_end = (void*)(long)ctx->data_end;
    void* data = (void*)(long)ctx->data;

    struct ethhdr* eth = data;
    if ((void*)(eth + 1) > data_end) {
        return XDP_DROP;
    }

    // 交换MAC地址
    unsigned char tmp[ETH_ALEN];
    __builtin_memcpy(tmp, eth->h_source, ETH_ALEN);
    __builtin_memcpy(eth->h_source, eth->h_dest, ETH_ALEN);
    __builtin_memcpy(eth->h_dest, tmp, ETH_ALEN);

    // 原路返回
    return XDP_TX;
}
```

```cpp
// 一致性哈希实现负载均衡

#include <map>
#include <string>
#include <vector>
#include <functional>

class ConsistentHash {
private:
    std::map<uint32_t, std::string> ring_;
    std::hash<std::string> hasher_;
    int virtual_nodes_;  // 虚拟节点数

    uint32_t hash(const std::string& key) {
        return hasher_(key);
    }

public:
    explicit ConsistentHash(int virtual_nodes = 150)
        : virtual_nodes_(virtual_nodes) {}

    // 添加节点
    void add_node(const std::string& node) {
        for (int i = 0; i < virtual_nodes_; i++) {
            std::string vnode = node + "#" + std::to_string(i);
            uint32_t hash_val = hash(vnode);
            ring_[hash_val] = node;
        }
    }

    // 移除节点
    void remove_node(const std::string& node) {
        for (int i = 0; i < virtual_nodes_; i++) {
            std::string vnode = node + "#" + std::to_string(i);
            uint32_t hash_val = hash(vnode);
            ring_.erase(hash_val);
        }
    }

    // 查找key对应的节点
    std::string get_node(const std::string& key) {
        if (ring_.empty()) {
            return "";
        }

        uint32_t hash_val = hash(key);

        // 在环上顺时针查找第一个节点
        auto it = ring_.lower_bound(hash_val);

        if (it == ring_.end()) {
            // 回绕到第一个节点
            it = ring_.begin();
        }

        return it->second;
    }

    // 获取多个副本节点
    std::vector<std::string> get_nodes(const std::string& key, int count) {
        std::vector<std::string> result;
        if (ring_.empty()) {
            return result;
        }

        uint32_t hash_val = hash(key);
        auto it = ring_.lower_bound(hash_val);

        std::set<std::string> unique_nodes;

        while (unique_nodes.size() < count && unique_nodes.size() < ring_.size()) {
            if (it == ring_.end()) {
                it = ring_.begin();
            }

            unique_nodes.insert(it->second);
            ++it;
        }

        result.assign(unique_nodes.begin(), unique_nodes.end());
        return result;
    }

    // 统计节点负载分布
    std::map<std::string, int> get_distribution(
            const std::vector<std::string>& keys) {
        std::map<std::string, int> dist;

        for (const auto& key : keys) {
            std::string node = get_node(key);
            dist[node]++;
        }

        return dist;
    }
};
```

```cpp
#include <hiredis/hiredis.h>
#include <vector>
#include <future>

class RedisClusterClient {
private:
    struct Node {
        redisContext* ctx;
        std::string host;
        int port;
    };

    std::vector<Node> nodes_;
    ConsistentHash hash_ring_;

public:
    // 初始化集群连接
    void Init(const std::vector<std::pair<std::string, int>>& servers) {
        for (const auto& [host, port] : servers) {
            // 连接优化
            struct timeval timeout = {1, 500000};  // 1.5秒超时
            redisContext* ctx = redisConnectWithTimeout(host.c_str(),
                                                       port, timeout);

            if (ctx == nullptr || ctx->err) {
                continue;
            }

            // 启用TCP_NODELAY
            int flag = 1;
            setsockopt(ctx->fd, IPPROTO_TCP, TCP_NODELAY,
                      &flag, sizeof(flag));

            // 设置TCP keep-alive
            flag = 1;
            setsockopt(ctx->fd, SOL_SOCKET, SO_KEEPALIVE,
                      &flag, sizeof(flag));

            nodes_.push_back({ctx, host, port});

            // 添加到一致性哈希环
            std::string node_id = host + ":" + std::to_string(port);
            hash_ring_.add_node(node_id);
        }
    }

    // Pipeline批量操作
    std::vector<std::string> BatchGet(const std::vector<std::string>& keys) {
        std::map<std::string, std::vector<std::string>> node_keys;

        // 按节点分组
        for (const auto& key : keys) {
            std::string node = hash_ring_.get_node(key);
            node_keys[node].push_back(key);
        }

        std::vector<std::future<std::vector<std::string>>> futures;

        // 并发查询多个节点
        for (const auto& [node, keys] : node_keys) {
            futures.push_back(std::async(std::launch::async,
                [this, &keys, &node]() {
                    return PipelineGet(GetContext(node), keys);
                }));
        }

        // 汇总结果
        std::vector<std::string> results;
        for (auto& fut : futures) {
            auto partial = fut.get();
            results.insert(results.end(), partial.begin(), partial.end());
        }

        return results;
    }

private:
    redisContext* GetContext(const std::string& node_id) {
        for (auto& node : nodes_) {
            std::string id = node.host + ":" + std::to_std::to_string(node.port);
            if (id == node_id) {
                return node.ctx;
            }
        }
        return nullptr;
    }

    std::vector<std::string> PipelineGet(redisContext* ctx,
                                        const std::vector<std::string>& keys) {
        // Pipeline: 批量发送命令
        for (const auto& key : keys) {
            redisAppendCommand(ctx, "GET %s", key.c_str());
        }

        // 批量接收响应
        std::vector<std::string> results;
        for (size_t i = 0; i < keys.size(); i++) {
            redisReply* reply;
            redisGetReply(ctx, (void**)&reply);

            if (reply->type == REDIS_REPLY_STRING) {
                results.emplace_back(reply->str, reply->len);
            } else {
                results.emplace_back("");
            }

            freeReplyObject(reply);
        }

        return results;
    }
};

// 客户端缓存（减少网络往返）
class ClientSideCache {
private:
    std::unordered_map<std::string, std::pair<std::string, time_t>> cache_;
    std::mutex mutex_;
    int ttl_seconds_;

public:
    explicit ClientSideCache(int ttl = 60) : ttl_seconds_(ttl) {}

    std::optional<std::string> Get(const std::string& key) {
        std::lock_guard<std::mutex> lock(mutex_);

        auto it = cache_.find(key);
        if (it != cache_.end()) {
            auto& [value, expire_time] = it->second;

            if (time(nullptr) < expire_time) {
                return value;  // 命中缓存
            } else {
                cache_.erase(it);  // 过期
            }
        }

        return std::nullopt;
    }

    void Set(const std::string& key, const std::string& value) {
        std::lock_guard<std::mutex> lock(mutex_);
        time_t expire = time(nullptr) + ttl_seconds_;
        cache_[key] = {value, expire};
    }

    // LRU淘汰（简化版）
    void EvictLRU(size_t max_size) {
        std::lock_guard<std::mutex> lock(mutex_);

        if (cache_.size() > max_size) {
            // 简单实现：删除最早过期的
            time_t min_expire = time(nullptr) + ttl_seconds_;
            std::string key_to_evict;

            for (const auto& [key, pair] : cache_) {
                if (pair.second < min_expire) {
                    min_expire = pair.second;
                    key_to_evict = key;
                }
            }

            if (!key_to_evict.empty()) {
                cache_.erase(key_to_evict);
            }
        }
    }
};
```

```sql
CREATE INDEX idx_order_date ON orders(created_at);

CREATE INDEX idx_order_user_date ON orders(user_id, created_at DESC);

-- 查询自动使用复合索引
SELECT * FROM orders
WHERE user_id = 123
  AND created_at > '2024-01-01'

CREATE INDEX idx_users_email_cover ON users(email, user_id, name);
```

```python
import torch
import torch.nn as nn
import onnx
import onnxruntime as ort
import numpy as np

class ModelOptimizer:
    def __init__(self, model):
        self.model = model

    # 1. 模型量化（INT8）
    def quantize_model(self):
        """
        量化：FP32 → INT8
        性能提升：2-4x
        模型大小：减少75%
        精度损失：<1%
        """
        # 动态量化（简单）
        quantized_model = torch.quantization.quantize_dynamic(
            self.model,
            {nn.Linear, nn.Conv2d},
            dtype=torch.qint8
        )

        return quantized_model

    # 2. 模型剪枝
    def prune_model(self, amount=0.3):
        """
        剪枝：移除不重要的权重
        性能提升：1.5-2x
        精度损失：<2%
        """
        import torch.nn.utils.prune as prune

        for name, module in self.model.named_modules():
            if isinstance(module, nn.Conv2d):
                prune.l1_unstructured(module, name='weight', amount=amount)
                prune.remove(module, 'weight')

        return self.model

    # 3. ONNX导出
    def export_onnx(self, dummy_input, output_path="model.onnx"):
        """
        导出ONNX格式
        跨平台部署
        支持多种运行时优化
        """
        torch.onnx.export(
            self.model,
            dummy_input,
            output_path,
            export_params=True,
            opset_version=15,
            do_constant_folding=True,  # 常量折叠优化
            input_names=['input'],
            output_names=['output'],
            dynamic_axes={
                'input': {0: 'batch_size'},
                'output': {0: 'batch_size'}
            }
        )

    # 4. TensorRT优化（NVIDIA GPU）
    def optimize_tensorrt(self, onnx_path):
        """
        TensorRT优化
        性能提升：5-10x
        需要NVIDIA GPU
        """
        import tensorrt as trt

        logger = trt.Logger(trt.Logger.WARNING)
        builder = trt.Builder(logger)
        network = builder.create_network(
            1 << int(trt.NetworkDefinitionCreationFlag.EXPLICIT_BATCH)
        )
        parser = trt.OnnxParser(network, logger)

        with open(onnx_path, 'rb') as model:
            parser.parse(model.read())

        config = builder.create_builder_config()
        config.max_workspace_size = 1 << 30  # 1GB

        # 启用FP16
        if builder.platform_has_fast_fp16:
            config.set_flag(trt.BuilderFlag.FP16)

        # 启用INT8（需要校准）
        if builder.platform_has_fast_int8:
            config.set_flag(trt.BuilderFlag.INT8)

        engine = builder.build_engine(network, config)
        return engine

    # 5. 批处理推理
    def batch_inference(self, inputs, batch_size=32):
        """
        批量推理
        性能提升：3-5x
        """
        results = []

        with torch.no_grad():
            for i in range(0, len(inputs), batch_size):
                batch = inputs[i:i+batch_size]
                batch_tensor = torch.stack(batch)

                # 批量推理
                output = self.model(batch_tensor)
                results.extend(output)

        return results

    # 6. 异步推理（多GPU）
    def multi_gpu_inference(self, inputs):
        """
        多GPU并行推理
        性能提升：线性扩展
        """
        from torch.nn import DataParallel

        # 数据并行
        parallel_model = DataParallel(self.model)

        with torch.no_grad():
            output = parallel_model(inputs)

        return output
```

```c
// 3. 并行编码
typedef struct {
    AVCodecContext* ctx;
    AVPacket* pkt;
    int frame_count;
} EncoderThread;

void* encoder_thread(void* arg) {
    EncoderThread* et = (EncoderThread*)arg;

    while (1) {
        AVFrame* frame = get_next_frame();
        if (!frame) break;

        avcodec_send_frame(et->ctx, frame);

        while (avcodec_receive_packet(et->ctx, et->pkt) == 0) {
            write_packet(et->pkt);
            av_packet_unref(et->pkt);
        }

        et->frame_count++;
    }

    return NULL;
}

void parallel_encode(int num_threads) {
    EncoderThread* threads = malloc(sizeof(EncoderThread) * num_threads);
    pthread_t* tids = malloc(sizeof(pthread_t) * num_threads);

    // 为每个线程创建独立的编码器
    for (int i = 0; i < num_threads; i++) {
        threads[i].ctx = create_encoder();
        threads[i].pkt = av_packet_alloc();
        threads[i].frame_count = 0;

        pthread_create(&tids[i], NULL, encoder_thread, &threads[i]);

    // 等待完成
    for (int i = 0; i < num_threads; i++) {
        pthread_join(tids[i], NULL);
        printf("Thread %d encoded %d frames\n", i, threads[i].frame_count);
    }

    // 线性扩展：4线程 ≈ 4x吞吐量
}
```
