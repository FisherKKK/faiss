### 1.1 高性能对象池

```cpp
// object_pool.hpp
// 线程安全的高性能对象池，避免频繁内存分配
#pragma once
#include <memory>
#include <vector>
#include <mutex>
#include <atomic>

template<typename T>
class ObjectPool {
private:
    struct alignas(64) Block {
        union {
            T object;
            Block* next;
        };
        bool in_use;
    };

    std::vector<std::unique_ptr<Block[]>> chunks_;
    std::atomic<Block*> free_list_{nullptr};
    std::mutex alloc_mutex_;
    size_t chunk_size_;

    void allocate_chunk() {
        auto chunk = std::make_unique<Block[]>(chunk_size_);

        // 将新chunk中的所有block链接到free_list
        for (size_t i = 0; i < chunk_size_ - 1; ++i) {
            chunk[i].next = &chunk[i + 1];
            chunk[i].in_use = false;
        }
        chunk[chunk_size_ - 1].next = free_list_.load(std::memory_order_relaxed);
        chunk[chunk_size_ - 1].in_use = false;

        free_list_.store(&chunk[0], std::memory_order_release);
        chunks_.push_back(std::move(chunk));
    }

public:
    explicit ObjectPool(size_t chunk_size = 1024)
        : chunk_size_(chunk_size) {
        std::lock_guard<std::mutex> lock(alloc_mutex_);
        allocate_chunk();
    }

    template<typename... Args>
    T* acquire(Args&&... args) {
        Block* block = free_list_.load(std::memory_order_acquire);

        while (block != nullptr) {
            Block* next = block->next;
            if (free_list_.compare_exchange_weak(block, next,
                    std::memory_order_release,
                    std::memory_order_acquire)) {
                block->in_use = true;
                new (&block->object) T(std::forward<Args>(args)...);
                return &block->object;
            }
        }

        // Free list为空，分配新chunk
        std::lock_guard<std::mutex> lock(alloc_mutex_);
        allocate_chunk();
        return acquire(std::forward<Args>(args)...);
    }

    void release(T* obj) {
        if (!obj) return;

        Block* block = reinterpret_cast<Block*>(obj);
        obj->~T();

        Block* old_head = free_list_.load(std::memory_order_relaxed);
        do {
            block->next = old_head;
        } while (!free_list_.compare_exchange_weak(old_head, block,
                    std::memory_order_release,
                    std::memory_order_relaxed));
    }

    // RAII包装器
    class Handle {
        ObjectPool* pool_;
        T* ptr_;
    public:
        Handle(ObjectPool* pool, T* ptr) : pool_(pool), ptr_(ptr) {}
        ~Handle() { if (ptr_) pool_->release(ptr_); }

        Handle(Handle&& other) noexcept
            : pool_(other.pool_), ptr_(other.ptr_) {
            other.ptr_ = nullptr;
        }

        Handle& operator=(Handle&& other) noexcept {
            if (this != &other) {
                if (ptr_) pool_->release(ptr_);
                pool_ = other.pool_;
                ptr_ = other.ptr_;
                other.ptr_ = nullptr;
            }
            return *this;
        }

        T* get() { return ptr_; }
        T* operator->() { return ptr_; }
        T& operator*() { return *ptr_; }
    };

    template<typename... Args>
    Handle acquire_handle(Args&&... args) {
        return Handle(this, acquire(std::forward<Args>(args)...));
    }
};

// 使用示例
/*
struct Connection {
    int fd;
    std::string buffer;
    Connection(int f) : fd(f) {}
};

void handle_request() {
    auto conn = conn_pool.acquire_handle(socket_fd);
    // 使用conn，退出作用域自动释放
    conn->buffer = "data";
}
```

### 1.2 Arena分配器

```cpp
// arena_allocator.hpp
// 快速内存分配器，适用于批量分配、统一释放的场景
#pragma once
#include <cstddef>
#include <memory>
#include <vector>

class Arena {
private:
    struct Block {
        std::unique_ptr<char[]> data;
        size_t size;
        size_t used;

        Block(size_t sz) : data(new char[sz]), size(sz), used(0) {}
    };

    std::vector<Block> blocks_;
    size_t block_size_;
    size_t total_allocated_{0};

    void allocate_block(size_t min_size) {
        size_t size = std::max(block_size_, min_size);
        blocks_.emplace_back(size);
        total_allocated_ += size;
    }

public:
    explicit Arena(size_t block_size = 4096)
        : block_size_(block_size) {
        allocate_block(block_size_);
    }

    // 禁止拷贝
    Arena(const Arena&) = delete;
    Arena& operator=(const Arena&) = delete;

    void* allocate(size_t size, size_t alignment = alignof(std::max_align_t)) {
        if (blocks_.empty()) {
            allocate_block(block_size_);
        }

        Block& current = blocks_.back();

        // 计算对齐后的位置
        size_t aligned_used = (current.used + alignment - 1) & ~(alignment - 1);

        if (aligned_used + size > current.size) {
            // 当前block不够，分配新block
            allocate_block(size);
            return allocate(size, alignment);
        }

        void* ptr = current.data.get() + aligned_used;
        current.used = aligned_used + size;
        return ptr;
    }

    template<typename T, typename... Args>
    T* create(Args&&... args) {
        void* mem = allocate(sizeof(T), alignof(T));
        return new (mem) T(std::forward<Args>(args)...);
    }

    template<typename T>
    T* allocate_array(size_t count) {
        return static_cast<T*>(allocate(sizeof(T) * count, alignof(T)));
    }

    void reset() {
        for (auto& block : blocks_) {
            block.used = 0;
        }
    }

    size_t total_allocated() const { return total_allocated_; }

    size_t total_used() const {
        size_t used = 0;
        for (const auto& block : blocks_) {
            used += block.used;
        }
        return used;
    }
};

// 使用示例
/*

// 批量创建对象

    auto* obj = arena.create<MyObject>(i);
    // 使用obj
}

// 统一释放所有对象（无需逐个delete）
arena.reset();
*/
```

```cpp
// numa_allocator.hpp
// NUMA架构下的感知型内存分配器
#pragma once
#include <numa.h>
#include <numaif.h>
#include <sys/mman.h>
#include <cstddef>

class NUMAAllocator {
private:
    int node_;
    bool use_hugepages_;

public:
    explicit NUMAAllocator(int node = -1, bool hugepages = true)
        : node_(node), use_hugepages_(hugepages) {
        if (node_ == -1) {
            node_ = numa_node_of_cpu(sched_getcpu());
        }
    }

    void* allocate(size_t size, bool lock_memory = true) {
        void* ptr;

        if (node_ >= 0 && node_ < numa_num_configured_nodes()) {
            // NUMA节点特定分配
            ptr = numa_alloc_onnode(size, node_);
        } else {
            // 本地节点分配
            ptr = numa_alloc_local(size);
        }

        if (!ptr) {
            throw std::bad_alloc();
        }

        // 启用大页
        if (use_hugepages_) {
            madvise(ptr, size, MADV_HUGEPAGE);
        }

        // First-touch策略：触发物理页分配
        memset(ptr, 0, size);

        // 锁定内存防止swap
        if (lock_memory) {
            if (mlock(ptr, size) != 0) {
                // 锁定失败不致命，继续使用
            }
        }

        return ptr;
    }

    void deallocate(void* ptr, size_t size) {
        if (ptr) {
            munlock(ptr, size);
            numa_free(ptr, size);
        }
    }

    // 获取内存所在的NUMA节点
    static int get_numa_node(void* ptr) {
        int node = -1;
        get_mempolicy(&node, nullptr, 0, ptr, MPOL_F_NODE | MPOL_F_ADDR);
        return node;
    }

    // 绑定当前线程到NUMA节点
    static void bind_thread_to_node(int node) {
        struct bitmask* mask = numa_allocate_cpumask();
        numa_node_to_cpus(node, mask);
        numa_sched_setaffinity(0, mask);
        numa_free_cpumask(mask);
    }

    // 设置内存访问策略
    static void set_memory_policy(void* ptr, size_t size, int node) {
        unsigned long nodemask = 1UL << node;
        mbind(ptr, size, MPOL_BIND, &nodemask, sizeof(nodemask) * 8,
              MPOL_MF_MOVE | MPOL_MF_STRICT);
    }
};

// STL兼容的NUMA分配器
template<typename T>
class STLNUMAAllocator {
private:
    NUMAAllocator allocator_;

public:
    using value_type = T;

    explicit STLNUMAAllocator(int node = -1) : allocator_(node) {}

    template<typename U>
    STLNUMAAllocator(const STLNUMAAllocator<U>& other)
        : allocator_(other.allocator_) {}

    T* allocate(size_t n) {
        return static_cast<T*>(allocator_.allocate(n * sizeof(T)));
    }

    void deallocate(T* ptr, size_t n) {
        allocator_.deallocate(ptr, n * sizeof(T));
    }

    template<typename U>
    struct rebind {
        using other = STLNUMAAllocator<U>;
    };
};
```

### 2.1 Lock-Free队列 (SPSC)

```cpp
// lock_free_queue.hpp
// 单生产者单消费者无锁队列（最高性能）
#pragma once
#include <atomic>
#include <cstddef>
#include <optional>

template<typename T, size_t Capacity>
class SPSCQueue {
    static_assert((Capacity & (Capacity - 1)) == 0,
                  "Capacity must be power of 2");
private:
    struct alignas(64) CacheLine {
        std::atomic<size_t> value{0};
        char padding[64 - sizeof(std::atomic<size_t>)];
    };

    CacheLine head_;  // Consumer owns
    CacheLine tail_;  // Producer owns
    T buffer_[Capacity];

public:
    SPSCQueue() = default;

    // Producer: 无锁入队
    bool try_enqueue(const T& item) {
        size_t tail = tail_.value.load(std::memory_order_relaxed);
        size_t next_tail = (tail + 1) & (Capacity - 1);

        size_t head = head_.value.load(std::memory_order_acquire);

        if (next_tail == head) {
            return false;  // Queue full
        }

        buffer_[tail] = item;
        tail_.value.store(next_tail, std::memory_order_release);
        return true;
    }

    bool try_enqueue(T&& item) {
        size_t tail = tail_.value.load(std::memory_order_relaxed);
        size_t next_tail = (tail + 1) & (Capacity - 1);

        size_t head = head_.value.load(std::memory_order_acquire);

        if (next_tail == head) {
            return false;
        }

        buffer_[tail] = std::move(item);
        tail_.value.store(next_tail, std::memory_order_release);
        return true;
    }

    // Consumer: 无锁出队
    std::optional<T> try_dequeue() {
        size_t head = head_.value.load(std::memory_order_relaxed);
        size_t tail = tail_.value.load(std::memory_order_acquire);

        if (head == tail) {
            return std::nullopt;  // Queue empty
        }

        T item = std::move(buffer_[head]);
        head_.value.store((head + 1) & (Capacity - 1),
                         std::memory_order_release);
        return item;
    }

    // 查询队列状态
    size_t size() const {
        size_t head = head_.value.load(std::memory_order_acquire);
        size_t tail = tail_.value.load(std::memory_order_acquire);
        return (tail - head) & (Capacity - 1);
    }

    bool empty() const {
        size_t head = head_.value.load(std::memory_order_acquire);
        size_t tail = tail_.value.load(std::memory_order_acquire);
        return head == tail;
    }

    size_t capacity() const { return Capacity; }
};

// 使用示例
/*

// Producer线程

    }
});
```

### 2.2 Lock-Free哈希表

```cpp
// lock_free_hashmap.hpp
// 多线程无锁哈希表
#pragma once
#include <atomic>
#include <functional>
#include <optional>
#include <memory>

template<typename K, typename V, size_t Capacity = 1024>
class LockFreeHashMap {
    static_assert((Capacity & (Capacity - 1)) == 0,
                  "Capacity must be power of 2");
private:
    struct Node {
        K key;
        V value;
        std::atomic<Node*> next{nullptr};

        Node(const K& k, const V& v) : key(k), value(v) {}
    };

    struct alignas(64) Bucket {
        std::atomic<Node*> head{nullptr};
    };

    Bucket buckets_[Capacity];
    std::hash<K> hasher_;

    size_t get_bucket(const K& key) const {
        return hasher_(key) & (Capacity - 1);
    }

public:
    LockFreeHashMap() = default;

    ~LockFreeHashMap() {
        for (size_t i = 0; i < Capacity; ++i) {
            Node* node = buckets_[i].head.load(std::memory_order_relaxed);
            while (node) {
                Node* next = node->next.load(std::memory_order_relaxed);
                delete node;
                node = next;
            }
        }
    }

    // 插入或更新
    bool insert_or_update(const K& key, const V& value) {
        size_t bucket_idx = get_bucket(key);
        Bucket& bucket = buckets_[bucket_idx];

        // 首先尝试查找并更新
        Node* node = bucket.head.load(std::memory_order_acquire);
        while (node) {
            if (node->key == key) {
                // 找到相同key，更新value
                // 注意：这里假设V的赋值是原子的或不需要原子性
                node->value = value;
                return true;
            }
            node = node->next.load(std::memory_order_acquire);
        }

        // 未找到，插入新节点
        Node* new_node = new Node(key, value);

        Node* old_head = bucket.head.load(std::memory_order_relaxed);
        do {
            new_node->next.store(old_head, std::memory_order_relaxed);
        } while (!bucket.head.compare_exchange_weak(
            old_head, new_node,
            std::memory_order_release,
            std::memory_order_relaxed));

        return true;
    }

    // 查找
    std::optional<V> find(const K& key) const {
        size_t bucket_idx = get_bucket(key);
        const Bucket& bucket = buckets_[bucket_idx];

        Node* node = bucket.head.load(std::memory_order_acquire);
        while (node) {
            if (node->key == key) {
                return node->value;
            }
            node = node->next.load(std::memory_order_acquire);
        }

        return std::nullopt;
    }

    // 删除（简化版本，真实实现需要考虑内存回收）
    bool erase(const K& key) {
        size_t bucket_idx = get_bucket(key);
        Bucket& bucket = buckets_[bucket_idx];

        Node** prev_ptr = reinterpret_cast<Node**>(&bucket.head);
        Node* node = bucket.head.load(std::memory_order_acquire);

        while (node) {
            if (node->key == key) {
                Node* next = node->next.load(std::memory_order_relaxed);

                // 注意：这里的CAS不是完全无锁的，真实实现需要使用hazard pointers
                Node* expected = node;
                if (reinterpret_cast<std::atomic<Node*>*>(prev_ptr)->
                    compare_exchange_strong(expected, next,
                        std::memory_order_release,
                        std::memory_order_relaxed)) {
                    delete node;  // 危险：可能有其他线程正在访问
                    return true;
                }
            }
            prev_ptr = reinterpret_cast<Node**>(&node->next);
            node = node->next.load(std::memory_order_acquire);
        }

        return false;
    }
};

// 使用示例
```

### 2.3 缓存友好的B+树

```cpp
// cache_friendly_btree.hpp
// 针对现代CPU缓存优化的B+树
#pragma once
#include <algorithm>
#include <cstring>
#include <vector>

template<typename K, typename V, size_t NodeSize = 64>
class CacheFriendlyBTree {
private:
    static constexpr size_t KEYS_PER_NODE =
        (NodeSize - sizeof(bool) - sizeof(void*)) / (sizeof(K) + sizeof(void*));

    struct alignas(NodeSize) Node {
        bool is_leaf;
        size_t num_keys;
        K keys[KEYS_PER_NODE];

        union {
            Node* children[KEYS_PER_NODE + 1];  // Internal node
            V values[KEYS_PER_NODE];             // Leaf node
        };

        Node* next;  // For leaf nodes,链接到下一个叶子节点

        Node(bool leaf) : is_leaf(leaf), num_keys(0), next(nullptr) {}
    };

    Node* root_;

    // 二分查找（针对小数组优化）
    size_t binary_search(const K* keys, size_t n, const K& key) const {
        if (n <= 8) {
            // 小数组使用线性搜索（更cache-friendly）
            for (size_t i = 0; i < n; ++i) {
                if (keys[i] >= key) return i;
            }
            return n;
        }

        // 大数组使用二分搜索
        size_t left = 0, right = n;
        while (left < right) {
            size_t mid = left + (right - left) / 2;
            if (keys[mid] < key) {
                left = mid + 1;
            } else {
                right = mid;
            }
        }
        return left;
    }

    // 插入到未满的叶子节点
    void insert_into_leaf(Node* leaf, const K& key, const V& value) {
        size_t pos = binary_search(leaf->keys, leaf->num_keys, key);

        // 移动元素为新key腾出空间
        if (pos < leaf->num_keys) {
            std::memmove(&leaf->keys[pos + 1], &leaf->keys[pos],
                        (leaf->num_keys - pos) * sizeof(K));
            std::memmove(&leaf->values[pos + 1], &leaf->values[pos],
                        (leaf->num_keys - pos) * sizeof(V));
        }

        leaf->keys[pos] = key;
        leaf->values[pos] = value;
        leaf->num_keys++;
    }

    Node* split_leaf(Node* leaf) {
        Node* new_leaf = new Node(true);
        size_t mid = KEYS_PER_NODE / 2;

        // 将后半部分移动到新节点
        new_leaf->num_keys = leaf->num_keys - mid;
        std::memcpy(new_leaf->keys, &leaf->keys[mid],
                   new_leaf->num_keys * sizeof(K));
        std::memcpy(new_leaf->values, &leaf->values[mid],
                   new_leaf->num_keys * sizeof(V));

        leaf->num_keys = mid;

        // 更新链表
        new_leaf->next = leaf->next;
        leaf->next = new_leaf;

        return new_leaf;
    }

public:
    CacheFriendlyBTree() : root_(new Node(true)) {}

    ~CacheFriendlyBTree() {
        // 简化的析构函数，实际需要递归删除所有节点
        delete root_;
    }

    void insert(const K& key, const V& value) {
        Node* leaf = root_;

        // 从根向下查找叶子节点
        while (!leaf->is_leaf) {
            size_t pos = binary_search(leaf->keys, leaf->num_keys, key);
            leaf = leaf->children[pos];
        }

        // 插入到叶子节点
        if (leaf->num_keys < KEYS_PER_NODE) {
            insert_into_leaf(leaf, key, value);
        } else {
            // 节点已满，需要分裂
            Node* new_leaf = split_leaf(leaf);

            if (key < new_leaf->keys[0]) {
                insert_into_leaf(leaf, key, value);
            } else {
                insert_into_leaf(new_leaf, key, value);
            }

            // 简化：假设根节点不会分裂
            // 实际实现需要处理根节点分裂的情况
        }
    }

    std::optional<V> find(const K& key) const {
        Node* node = root_;

        while (!node->is_leaf) {
            size_t pos = binary_search(node->keys, node->num_keys, key);
            node = node->children[pos];
        }

        // 在叶子节点中查找
        size_t pos = binary_search(node->keys, node->num_keys, key);
        if (pos < node->num_keys && node->keys[pos] == key) {
            return node->values[pos];
        }

        return std::nullopt;
    }

    // 范围查询（利用叶子节点链表）
    std::vector<std::pair<K, V>> range_query(const K& start, const K& end) const {
        std::vector<std::pair<K, V>> result;

        // 找到起始叶子节点
        Node* leaf = root_;
        while (!leaf->is_leaf) {
            size_t pos = binary_search(leaf->keys, leaf->num_keys, start);
            leaf = leaf->children[pos];
        }

        // 遍历叶子节点链表
        while (leaf) {
            for (size_t i = 0; i < leaf->num_keys; ++i) {
                if (leaf->keys[i] >= start && leaf->keys[i] < end) {
                    result.emplace_back(leaf->keys[i], leaf->values[i]);
                } else if (leaf->keys[i] >= end) {
                    return result;
                }
            }
            leaf = leaf->next;
        }

        return result;
    }
```

### 3.1 工作窃取线程池

```cpp
// work_stealing_threadpool.hpp
// 基于工作窃取算法的高性能线程池
#pragma once
#include <thread>
#include <vector>
#include <deque>
#include <mutex>
#include <condition_variable>
#include <functional>
#include <future>
#include <random>

class WorkStealingThreadPool {
private:
    struct WorkQueue {
        std::deque<std::function<void()>> tasks;
        std::mutex mutex;

        bool try_push(std::function<void()> task) {
            std::lock_guard<std::mutex> lock(mutex);
            tasks.push_back(std::move(task));
            return true;
        }

        bool try_pop(std::function<void()>& task) {
            std::lock_guard<std::mutex> lock(mutex);
            if (tasks.empty()) return false;
            task = std::move(tasks.front());
            tasks.pop_front();
            return true;
        }

        bool try_steal(std::function<void()>& task) {
            std::lock_guard<std::mutex> lock(mutex);
            if (tasks.empty()) return false;
            task = std::move(tasks.back());  // 从尾部窃取
            tasks.pop_back();
            return true;
        }

        size_t size() const {
            std::lock_guard<std::mutex> lock(const_cast<std::mutex&>(mutex));
            return tasks.size();
        }
    };

    std::vector<std::thread> workers_;
    std::vector<std::unique_ptr<WorkQueue>> queues_;
    std::atomic<bool> stopped_{false};
    std::atomic<size_t> active_workers_{0};

    static thread_local size_t thread_id_;
    static thread_local std::mt19937 rng_;

    void worker_thread(size_t id) {
        thread_id_ = id;
        rng_.seed(id);

        WorkQueue& my_queue = *queues_[id];

        while (!stopped_.load(std::memory_order_relaxed)) {
            std::function<void()> task;

            // 1. 尝试从自己的队列获取任务
            if (my_queue.try_pop(task)) {
                active_workers_.fetch_add(1, std::memory_order_relaxed);
                task();
                active_workers_.fetch_sub(1, std::memory_order_relaxed);
                continue;
            }

            // 2. 尝试从其他队列窃取任务
            bool stolen = false;
            for (size_t i = 0; i < queues_.size(); ++i) {
                size_t victim = (id + i + 1) % queues_.size();
                if (queues_[victim]->try_steal(task)) {
                    active_workers_.fetch_add(1, std::memory_order_relaxed);
                    task();
                    active_workers_.fetch_sub(1, std::memory_order_relaxed);
                    stolen = true;
                    break;
                }
            }

            if (!stolen) {
                // 3. 没有任务，yield让出CPU
                std::this_thread::yield();
            }
        }
    }

public:
    explicit WorkStealingThreadPool(size_t num_threads = 0) {
        if (num_threads == 0) {
            num_threads = std::thread::hardware_concurrency();
        }

        queues_.reserve(num_threads);
        for (size_t i = 0; i < num_threads; ++i) {
            queues_.push_back(std::make_unique<WorkQueue>());
        }

        workers_.reserve(num_threads);
        for (size_t i = 0; i < num_threads; ++i) {
            workers_.emplace_back(&WorkStealingThreadPool::worker_thread,
                                 this, i);
        }
    }

    ~WorkStealingThreadPool() {
        stopped_.store(true, std::memory_order_relaxed);
        for (auto& worker : workers_) {
            if (worker.joinable()) {
                worker.join();
            }
        }
    }

    template<typename F, typename... Args>
    auto submit(F&& f, Args&&... args)
        -> std::future<decltype(f(args...))> {
        using return_type = decltype(f(args...));

        auto task = std::make_shared<std::packaged_task<return_type()>>(
            std::bind(std::forward<F>(f), std::forward<Args>(args)...)
        );

        std::future<return_type> result = task->get_future();

        // 选择队列（轮询或随机）
        size_t queue_id;
        if (thread_id_ < queues_.size()) {
            // 在工作线程中，提交到自己的队列
            queue_id = thread_id_;
        } else {
            // 在外部线程中，随机选择队列
            queue_id = rng_() % queues_.size();
        }

        queues_[queue_id]->try_push([task]() { (*task)(); });

        return result;
    }

    size_t num_threads() const { return workers_.size(); }

    size_t active_tasks() const {
        return active_workers_.load(std::memory_order_relaxed);
    }

    size_t pending_tasks() const {
        size_t total = 0;
        for (const auto& queue : queues_) {
            total += queue->size();
        }
        return total;
    }
};

thread_local size_t WorkStealingThreadPool::thread_id_ = 0;
thread_local std::mt19937 WorkStealingThreadPool::rng_;
```

```cpp
// rcu_lock.hpp
// Read-Copy-Update风格的无锁读共享数据结构
#pragma once
#include <atomic>
#include <memory>
#include <functional>

template<typename T>
class RCUProtected {
private:
    std::atomic<T*> ptr_{nullptr};
    std::atomic<size_t> readers_{0};

public:
    explicit RCUProtected(T* initial = nullptr) : ptr_(initial) {}

    ~RCUProtected() {
        T* ptr = ptr_.load(std::memory_order_relaxed);
        delete ptr;
    }

    // 读操作：快速、无锁
    template<typename Func>
    auto read(Func&& func) -> decltype(func(std::declval<const T&>())) {
        readers_.fetch_add(1, std::memory_order_acquire);

        T* ptr = ptr_.load(std::memory_order_consume);
        if (!ptr) {
            readers_.fetch_sub(1, std::memory_order_release);
            throw std::runtime_error("Null pointer access");
        }

        auto result = func(*ptr);

        readers_.fetch_sub(1, std::memory_order_release);
        return result;
    }

    // 写操作：创建新副本
    template<typename Func>
    void update(Func&& func) {
        T* old_ptr = ptr_.load(std::memory_order_acquire);
        T* new_ptr = old_ptr ? new T(*old_ptr) : new T();

        func(*new_ptr);

        ptr_.store(new_ptr, std::memory_order_release);

        // 等待所有读者离开
        while (readers_.load(std::memory_order_acquire) > 0) {
            std::this_thread::yield();
        }

        delete old_ptr;
    }

    // 替换整个数据
    void replace(T* new_ptr) {
        T* old_ptr = ptr_.exchange(new_ptr, std::memory_order_acq_rel);

        while (readers_.load(std::memory_order_acquire) > 0) {
            std::this_thread::yield();
        }

        delete old_ptr;
    }
};
```

```cpp
// simd_operations.hpp
// AVX2 SIMD操作库
#pragma once
#include <immintrin.h>
#include <cstddef>
#include <cstring>

class SIMDOperations {
public:
    // 向量加法 (float)
    static void vector_add_f32(const float* a, const float* b,
                               float* result, size_t n) {
        size_t i = 0;

        // AVX2处理：每次8个float
        for (; i + 8 <= n; i += 8) {
            __m256 va = _mm256_loadu_ps(&a[i]);
            __m256 vb = _mm256_loadu_ps(&b[i]);
            __m256 vr = _mm256_add_ps(va, vb);
            _mm256_storeu_ps(&result[i], vr);
        }

        // 处理剩余元素
        for (; i < n; ++i) {
            result[i] = a[i] + b[i];
        }
    }

    // 点积 (float)
    static float dot_product_f32(const float* a, const float* b, size_t n) {
        __m256 sum_vec = _mm256_setzero_ps();
        size_t i = 0;

        // AVX2处理
        for (; i + 8 <= n; i += 8) {
            __m256 va = _mm256_loadu_ps(&a[i]);
            __m256 vb = _mm256_loadu_ps(&b[i]);
            sum_vec = _mm256_fmadd_ps(va, vb, sum_vec);  // FMA: a*b+sum
        }

        // 水平求和
        __m128 sum_high = _mm256_extractf128_ps(sum_vec, 1);
        __m128 sum_low = _mm256_castps256_ps128(sum_vec);
        __m128 sum128 = _mm_add_ps(sum_high, sum_low);

        sum128 = _mm_hadd_ps(sum128, sum128);
        sum128 = _mm_hadd_ps(sum128, sum128);

        float result;
        _mm_store_ss(&result, sum128);

        // 处理剩余元素
        for (; i < n; ++i) {
            result += a[i] * b[i];
        }

        return result;
    }

    // L2距离 (float)
    static float l2_distance_f32(const float* a, const float* b, size_t n) {
        __m256 sum_vec = _mm256_setzero_ps();
        size_t i = 0;

        for (; i + 8 <= n; i += 8) {
            __m256 va = _mm256_loadu_ps(&a[i]);
            __m256 vb = _mm256_loadu_ps(&b[i]);
            __m256 diff = _mm256_sub_ps(va, vb);
            sum_vec = _mm256_fmadd_ps(diff, diff, sum_vec);
        }

        // 水平求和
        __m128 sum_high = _mm256_extractf128_ps(sum_vec, 1);
        __m128 sum_low = _mm256_castps256_ps128(sum_vec);
        __m128 sum128 = _mm_add_ps(sum_high, sum_low);

        sum128 = _mm_hadd_ps(sum128, sum128);
        sum128 = _mm_hadd_ps(sum128, sum128);

        float result;
        _mm_store_ss(&result, sum128);

        for (; i < n; ++i) {
            float diff = a[i] - b[i];
            result += diff * diff;
        }

        return std::sqrt(result);
    }

    // 矩阵乘法 (简化版，仅支持小矩阵)
    static void matmul_f32(const float* A, const float* B, float* C,
                           size_t M, size_t N, size_t K) {
        // C[M×N] = A[M×K] × B[K×N]
        // 行优先存储

        for (size_t i = 0; i < M; ++i) {
            for (size_t j = 0; j < N; ++j) {
                __m256 sum = _mm256_setzero_ps();
                size_t k = 0;

                // SIMD内循环
                for (; k + 8 <= K; k += 8) {
                    __m256 a = _mm256_loadu_ps(&A[i * K + k]);
                    __m256 b = _mm256_loadu_ps(&B[k * N + j]);  // 需要转置B
                    sum = _mm256_fmadd_ps(a, b, sum);
                }

                // 求和
                float temp[8];
                _mm256_storeu_ps(temp, sum);
                float result = temp[0] + temp[1] + temp[2] + temp[3] +
                               temp[4] + temp[5] + temp[6] + temp[7];

                // 剩余元素
                for (; k < K; ++k) {
                    result += A[i * K + k] * B[k * N + j];
                }

                C[i * N + j] = result;
            }
        }
    }

    // 批量ReLU激活函数
    static void relu_f32(float* data, size_t n) {
        __m256 zero = _mm256_setzero_ps();
        size_t i = 0;

        for (; i + 8 <= n; i += 8) {
            __m256 v = _mm256_loadu_ps(&data[i]);
            v = _mm256_max_ps(v, zero);  // max(v, 0)
            _mm256_storeu_ps(&data[i], v);
        }

        for (; i < n; ++i) {
            data[i] = std::max(data[i], 0.0f);
        }
    }
};
```

```cpp
#define simd_load_f32 vld1q_f32

#define NO_SIMD
#endif

template<typename T>
class SIMDVector {
private:
    T* data_;
    size_t size_;
    bool owned_;

public:
    SIMDVector(size_t size)
        : size_(size), owned_(true) {
#ifdef __AVX2__
        data_ = static_cast<T*>(aligned_alloc(32, size * sizeof(T)));
#else
        data_ = new T[size];
#endif
    }

    SIMDVector(T* data, size_t size)
        : data_(data), size_(size), owned_(false) {}

    ~SIMDVector() {
        if (owned_) {
#ifdef __AVX2__
            free(data_);
#else
            delete[] data_;
#endif
        }
    }

    // 向量加法
    void add(const SIMDVector& other, SIMDVector& result) const {
#ifndef NO_SIMD
        size_t i = 0;
        for (; i + SIMD_WIDTH <= size_; i += SIMD_WIDTH) {
            simd_f32 a = simd_load_f32(&data_[i]);
            simd_f32 b = simd_load_f32(&other.data_[i]);
            simd_f32 r = simd_add_f32(a, b);
            simd_store_f32(&result.data_[i], r);
        }

        for (; i < size_; ++i) {
            result.data_[i] = data_[i] + other.data_[i];
        }
#else
        for (size_t i = 0; i < size_; ++i) {
            result.data_[i] = data_[i] + other.data_[i];
        }
#endif
    }

    // 点积

#ifndef NO_SIMD
        simd_f32 sum_vec = simd_setzero_f32();
        size_t i = 0;

        for (; i + SIMD_WIDTH <= size_; i += SIMD_WIDTH) {
            simd_f32 a = simd_load_f32(&data_[i]);
            simd_f32 b = simd_load_f32(&other.data_[i]);
            sum_vec = simd_fmadd_f32(a, b, sum_vec);

        for (int j = 0; j < SIMD_WIDTH; ++j) result += temp[j];
#elif defined(__ARM_NEON)

#endif

            result += data_[i] * other.data_[i];

#else

        for (size_t i = 0; i < size_; ++i) {
            result += data_[i] * other.data_[i];
        }
        return result;
#endif
    }

    T* data() { return data_; }
    const T* data() const { return data_; }
    size_t size() const { return size_; }
```

```cpp
// io_uring_wrapper.hpp
// Linux io_uring高性能异步IO封装
#pragma once
#include <liburing.h>
#include <functional>
#include <vector>
#include <memory>
#include <cstring>

class IoUringQueue {
private:
    struct io_uring ring_;
    bool initialized_{false};

    struct Request {
        std::function<void(ssize_t)> callback;
        void* buffer;
        size_t buffer_size;
        int fd;
    };

    std::vector<std::unique_ptr<Request>> pending_requests_;

public:
    explicit IoUringQueue(unsigned queue_depth = 256) {
        int ret = io_uring_queue_init(queue_depth, &ring_, 0);
        if (ret < 0) {
            throw std::runtime_error("io_uring_queue_init failed");
        }
        initialized_ = true;
    }

    ~IoUringQueue() {
        if (initialized_) {
            io_uring_queue_exit(&ring_);
        }
    }

    // 异步读
    void async_read(int fd, void* buffer, size_t size, off_t offset,
                    std::function<void(ssize_t)> callback) {
        struct io_uring_sqe* sqe = io_uring_get_sqe(&ring_);
        if (!sqe) {
            throw std::runtime_error("Failed to get SQE");
        }

        auto req = std::make_unique<Request>();
        req->callback = std::move(callback);
        req->buffer = buffer;
        req->buffer_size = size;
        req->fd = fd;

        io_uring_prep_read(sqe, fd, buffer, size, offset);
        io_uring_sqe_set_data(sqe, req.get());

        pending_requests_.push_back(std::move(req));
    }

    // 异步写
    void async_write(int fd, const void* buffer, size_t size, off_t offset,
                     std::function<void(ssize_t)> callback) {
        struct io_uring_sqe* sqe = io_uring_get_sqe(&ring_);
        if (!sqe) {
            throw std::runtime_error("Failed to get SQE");
        }

        auto req = std::make_unique<Request>();
        req->callback = std::move(callback);
        req->buffer = const_cast<void*>(buffer);
        req->buffer_size = size;
        req->fd = fd;

        io_uring_prep_write(sqe, fd, buffer, size, offset);
        io_uring_sqe_set_data(sqe, req.get());

        pending_requests_.push_back(std::move(req));
    }

    // 提交所有pending的请求
    int submit() {
        return io_uring_submit(&ring_);
    }

    // 处理完成的请求
    int process_completions(int wait_nr = 1) {
        struct io_uring_cqe* cqe;
        int processed = 0;

        while (true) {
            int ret = io_uring_wait_cqe(&ring_, &cqe);
            if (ret < 0) {
                if (ret == -EAGAIN || ret == -EINTR) {
                    break;
                }
                throw std::runtime_error("io_uring_wait_cqe failed");
            }

            Request* req = static_cast<Request*>(io_uring_cqe_get_data(cqe));
            if (req && req->callback) {
                req->callback(cqe->res);
            }

            io_uring_cqe_seen(&ring_, cqe);
            processed++;

            // 清理完成的请求
            auto it = std::find_if(pending_requests_.begin(),
                                   pending_requests_.end(),
                                   [req](const auto& r) { return r.get() == req; });
            if (it != pending_requests_.end()) {
                pending_requests_.erase(it);
            }

            if (processed >= wait_nr) break;
        }

        return processed;
    }

    // 批量读取
    void batch_read(int fd, std::vector<std::pair<void*, size_t>>& buffers,
                    off_t start_offset,
                    std::function<void(size_t, ssize_t)> callback) {
        off_t offset = start_offset;
        for (size_t i = 0; i < buffers.size(); ++i) {
            async_read(fd, buffers[i].first, buffers[i].second, offset,
                      [callback, i](ssize_t bytes) {
                          callback(i, bytes);
                      });
            offset += buffers[i].second;
        }
        submit();
    }
};
```

```cpp
// mmap_file.hpp
// 高性能内存映射文件操作
#pragma once
#include <sys/mman.h>
#include <sys/stat.h>
#include <fcntl.h>
#include <unistd.h>
#include <stdexcept>
#include <cstring>

class MMapFile {
private:
    int fd_{-1};
    void* data_{nullptr};
    size_t size_{0};
    bool read_only_{true};

public:
    MMapFile() = default;

    // 打开文件进行映射
    bool open(const char* filename, bool read_only = true, bool use_huge_pages = false) {
        read_only_ = read_only;

        int flags = read_only ? O_RDONLY : O_RDWR;
        fd_ = ::open(filename, flags);
        if (fd_ < 0) {
            return false;
        }

        struct stat st;
        if (fstat(fd_, &st) < 0) {
            close();
            return false;
        }
        size_ = st.st_size;

        int prot = read_only ? PROT_READ : (PROT_READ | PROT_WRITE);
        int mmap_flags = MAP_SHARED;

        if (use_huge_pages) {
            mmap_flags |= MAP_HUGETLB;
        }

        data_ = mmap(nullptr, size_, prot, mmap_flags, fd_, 0);
        if (data_ == MAP_FAILED) {
            data_ = nullptr;
            close();
            return false;
        }

        // 优化提示
        madvise(data_, size_, MADV_SEQUENTIAL);  // 顺序访问
        // madvise(data_, size_, MADV_RANDOM);   // 随机访问
        madvise(data_, size_, MADV_WILLNEED);    // 预读

        return true;
    }

    // 创建新的映射文件
    bool create(const char* filename, size_t size, bool use_huge_pages = false) {
        fd_ = ::open(filename, O_RDWR | O_CREAT | O_TRUNC, 0644);
        if (fd_ < 0) {
            return false;
        }

        // 扩展文件大小
        if (ftruncate(fd_, size) < 0) {
            close();
            return false;
        }

        size_ = size;
        read_only_ = false;

        int mmap_flags = MAP_SHARED;
        if (use_huge_pages) {
            mmap_flags |= MAP_HUGETLB;
        }

        data_ = mmap(nullptr, size_, PROT_READ | PROT_WRITE, mmap_flags, fd_, 0);
        if (data_ == MAP_FAILED) {
            data_ = nullptr;
            close();
            return false;
        }

        return true;
    }

    void close() {
        if (data_ != nullptr) {
            munmap(data_, size_);
            data_ = nullptr;
        }
        if (fd_ >= 0) {
            ::close(fd_);
            fd_ = -1;
        }
        size_ = 0;
    }

    ~MMapFile() {
        close();
    }

    // 同步到磁盘
    bool sync(bool async = false) {
        if (data_ == nullptr) return false;
        int flags = async ? MS_ASYNC : MS_SYNC;
        return msync(data_, size_, flags) == 0;
    }

    // 预读指定范围
    void prefetch(size_t offset, size_t length) {
        if (offset + length > size_) {
            length = size_ - offset;
        }
        madvise(static_cast<char*>(data_) + offset, length, MADV_WILLNEED);
    }

    // 告知不再需要指定范围
    void dontneed(size_t offset, size_t length) {
        if (offset + length > size_) {
            length = size_ - offset;
        }
        madvise(static_cast<char*>(data_) + offset, length, MADV_DONTNEED);
    }

    // 锁定内存（防止swap）
    bool lock_memory(size_t offset = 0, size_t length = 0) {
        if (length == 0) length = size_;
        return mlock(static_cast<char*>(data_) + offset, length) == 0;
    }

    void* data() { return data_; }
    const void* data() const { return data_; }
    size_t size() const { return size_; }
    bool is_open() const { return data_ != nullptr; }
    bool is_read_only() const { return read_only_; }

    // 类型安全访问
    template<typename T>
    T* as() { return static_cast<T*>(data_); }

    template<typename T>
    const T* as() const { return static_cast<const T*>(data_); }
};
```

```cpp
// cache_aligned.hpp
// 缓存行对齐的数据结构，避免false sharing
#pragma once
#include <cstddef>
#include <new>

// 缓存行大小常量
constexpr size_t CACHELINE_SIZE = 64;

// 缓存行对齐的包装器
template<typename T>
struct alignas(CACHELINE_SIZE) CacheAligned {
    T value;

    CacheAligned() = default;

    CacheAligned(const T& v) : value(v) {}

    CacheAligned(T&& v) : value(std::move(v)) {}

    operator T&() { return value; }
    operator const T&() const { return value; }

    T& get() { return value; }
    const T& get() const { return value; }
};

// 避免false sharing的计数器
template<typename T>
struct PaddedCounter {
    alignas(CACHELINE_SIZE) T value{0};
    char padding[CACHELINE_SIZE - sizeof(T)];

    PaddedCounter() = default;

    T fetch_add(T delta) {
        T old = value;
        value += delta;
        return old;
    }

    T load() const { return value; }
    void store(T val) { value = val; }
};

// 线程本地计数器（避免缓存竞争）
template<typename T>
class ThreadLocalCounter {
private:
    static constexpr size_t MAX_THREADS = 128;
    PaddedCounter<T> counters_[MAX_THREADS];

public:
    ThreadLocalCounter() = default;

    void add(size_t thread_id, T delta) {
        if (thread_id < MAX_THREADS) {
            counters_[thread_id].value += delta;
        }
    }

    T total() const {
        T sum = 0;
        for (size_t i = 0; i < MAX_THREADS; ++i) {
            sum += counters_[i].value;
        }
        return sum;
    }

    void reset() {
        for (size_t i = 0; i < MAX_THREADS; ++i) {
            counters_[i].value = 0;
        }
    }
};
```

```cpp
// 软件预取优化工具
#pragma once
#include <cstddef>

// GCC/Clang预取指令
#if defined(__GNUC__) || defined(__clang__)
    #define PREFETCH_READ(addr) __builtin_prefetch((addr), 0, 3)
    #define PREFETCH_WRITE(addr) __builtin_prefetch((addr), 1, 3)
    #define PREFETCH_TEMPORAL(addr) __builtin_prefetch((addr), 0, 2)
    #define PREFETCH_NON_TEMPORAL(addr) __builtin_prefetch((addr), 0, 0)
#else
    #define PREFETCH_READ(addr)
    #define PREFETCH_WRITE(addr)
    #define PREFETCH_TEMPORAL(addr)
    #define PREFETCH_NON_TEMPORAL(addr)
#endif

// 预取策略
class PrefetchStrategy {
public:
    // 线性预取（适用于顺序访问）
    template<typename T>
    static void linear_prefetch(const T* data, size_t current_index,
                                size_t prefetch_distance, size_t total_size) {
        size_t prefetch_index = current_index + prefetch_distance;
        if (prefetch_index < total_size) {
            PREFETCH_READ(&data[prefetch_index]);
        }
    }

    // 批量预取
    template<typename T>
    static void batch_prefetch(const T* data, const size_t* indices,
                              size_t count) {
        for (size_t i = 0; i < count; ++i) {
            PREFETCH_READ(&data[indices[i]]);
        }
    }

    // 间隔预取（适用于链表等数据结构）
    template<typename Node>
    static void strided_prefetch(const Node* node, size_t stride_count) {
        const Node* current = node;
        for (size_t i = 0; i < stride_count && current != nullptr; ++i) {
            PREFETCH_READ(current);
            current = current->next;
        }
    }
};

// 预取优化的遍历
template<typename T, typename Func>
void prefetch_loop(const T* data, size_t size, Func&& func,
                   size_t prefetch_distance = 8) {
    // 预取前几个元素
    for (size_t i = 0; i < std::min(prefetch_distance, size); ++i) {
        PREFETCH_READ(&data[i]);
    }

    // 主循环：处理当前元素，预取未来元素
    for (size_t i = 0; i < size; ++i) {
        if (i + prefetch_distance < size) {
            PREFETCH_READ(&data[i + prefetch_distance]);
        }
        func(data[i], i);
    }
}
```

```cpp
// high_performance_vector_db.hpp
// 简化的高性能向量数据库实现
#pragma once
#include <vector>
#include <memory>
#include <algorithm>
#include <queue>
#include <immintrin.h>

template<size_t DIM>
class VectorDatabase {
public:
    static_assert(DIM % 8 == 0, "Dimension must be multiple of 8 for AVX2");

    struct Vector {
        alignas(32) float data[DIM];
        int64_t id;

        Vector() : id(-1) {
            std::fill(data, data + DIM, 0.0f);
        }

        Vector(int64_t id_, const float* values) : id(id_) {
            std::memcpy(data, values, DIM * sizeof(float));
        }
    };

    struct SearchResult {
        int64_t id;
        float distance;

        bool operator<(const SearchResult& other) const {
            return distance > other.distance;  // Min heap
        }
    };

private:
    std::vector<Vector> vectors_;
    size_t max_vectors_;

    // SIMD优化的L2距离计算
    float l2_distance_simd(const float* a, const float* b) const {
        __m256 sum = _mm256_setzero_ps();

        for (size_t i = 0; i < DIM; i += 8) {
            __m256 va = _mm256_load_ps(&a[i]);
            __m256 vb = _mm256_load_ps(&b[i]);
            __m256 diff = _mm256_sub_ps(va, vb);
            sum = _mm256_fmadd_ps(diff, diff, sum);
        }

        // 水平求和
        __m128 sum_high = _mm256_extractf128_ps(sum, 1);
        __m128 sum_low = _mm256_castps256_ps128(sum);
        __m128 sum128 = _mm_add_ps(sum_high, sum_low);

        sum128 = _mm_hadd_ps(sum128, sum128);
        sum128 = _mm_hadd_ps(sum128, sum128);

        float result;
        _mm_store_ss(&result, sum128);

        return std::sqrt(result);
    }

    // SIMD优化的内积计算
    float dot_product_simd(const float* a, const float* b) const {
        __m256 sum = _mm256_setzero_ps();

        for (size_t i = 0; i < DIM; i += 8) {
            __m256 va = _mm256_load_ps(&a[i]);
            __m256 vb = _mm256_load_ps(&b[i]);
            sum = _mm256_fmadd_ps(va, vb, sum);
        }

        __m128 sum_high = _mm256_extractf128_ps(sum, 1);
        __m128 sum_low = _mm256_castps256_ps128(sum);
        __m128 sum128 = _mm_add_ps(sum_high, sum_low);

        sum128 = _mm_hadd_ps(sum128, sum128);
        sum128 = _mm_hadd_ps(sum128, sum128);

        float result;
        _mm_store_ss(&result, sum128);

        return result;
    }

public:
    explicit VectorDatabase(size_t max_vectors = 1000000)
        : max_vectors_(max_vectors) {
        vectors_.reserve(max_vectors);
    }

    // 添加向量
    int64_t add_vector(const float* data) {
        if (vectors_.size() >= max_vectors_) {
            return -1;
        }

        int64_t id = vectors_.size();
        vectors_.emplace_back(id, data);
        return id;
    }

    // 批量添加
    void add_vectors_batch(const float* data, size_t count) {
        for (size_t i = 0; i < count; ++i) {
            add_vector(&data[i * DIM]);
        }
    }

    // K近邻搜索 (暴力搜索 + SIMD优化)
    std::vector<SearchResult> search_knn(const float* query, size_t k) const {
        std::priority_queue<SearchResult> heap;

        for (const auto& vec : vectors_) {
            float dist = l2_distance_simd(query, vec.data);

            if (heap.size() < k) {
                heap.push({vec.id, dist});
            } else if (dist < heap.top().distance) {
                heap.pop();
                heap.push({vec.id, dist});
            }
        }

        std::vector<SearchResult> results;
        results.reserve(k);
        while (!heap.empty()) {
            results.push_back(heap.top());
            heap.pop();
        }

        std::reverse(results.begin(), results.end());
        return results;
    }

    // 余弦相似度搜索
    std::vector<SearchResult> search_cosine(const float* query, size_t k) const {
        std::priority_queue<SearchResult> heap;

        // 归一化query
        float query_norm = std::sqrt(dot_product_simd(query, query));
        alignas(32) float normalized_query[DIM];
        for (size_t i = 0; i < DIM; ++i) {
            normalized_query[i] = query[i] / query_norm;
        }

        for (const auto& vec : vectors_) {
            float dot = dot_product_simd(normalized_query, vec.data);
            float distance = 1.0f - dot;  // 余弦距离 = 1 - 余弦相似度

            if (heap.size() < k) {
                heap.push({vec.id, distance});
            } else if (distance < heap.top().distance) {
                heap.pop();
                heap.push({vec.id, distance});
            }
        }

        std::vector<SearchResult> results;
        results.reserve(k);
        while (!heap.empty()) {
            results.push_back(heap.top());
            heap.pop();
        }

        std::reverse(results.begin(), results.end());
        return results;
    }

    size_t size() const { return vectors_.size(); }

    const Vector& get_vector(size_t index) const {
        return vectors_[index];
    }
};
```
