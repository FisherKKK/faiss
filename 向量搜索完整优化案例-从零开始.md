### 1.2 基础实现：暴力搜索

```cpp
// 版本 1：朴素实现
class NaiveVectorSearch : public VectorSearch {
    std::vector<float> vectors;  // [n, d] 行主序存储
    size_t dimension;
    size_t num_vectors;

public:
    NaiveVectorSearch(size_t d) : dimension(d), num_vectors(0) {}

    void add_vectors(const float* vectors, size_t count) override {
        this->vectors.insert(
            this->vectors.end(),
            vectors,
            vectors + count * dimension
        );
        num_vectors += count;
    }

    void search(
        const float* query,
        size_t k,
        float* distances,
        size_t* labels) const override {

        // 计算所有距离
        std::vector<std::pair<float, size_t>> all_distances;
        all_distances.reserve(num_vectors);

        auto start = std::chrono::high_resolution_clock::now();

        for (size_t i = 0; i < num_vectors; i++) {
            // L2 距离
            float dist = 0;
            for (size_t j = 0; j < dimension; j++) {
                float diff = query[j] - vectors[i * dimension + j];
                dist += diff * diff;
            }
            all_distances.push_back({dist, i});
        }

        // 排序
        std::sort(all_distances.begin(), all_distances.end());

        // 返回 Top-K
        for (size_t i = 0; i < k; i++) {
            distances[i] = all_distances[i].first;
            labels[i] = all_distances[i].second;
        }
    }

    size_t size() const override {
        return num_vectors;
    }
};
```

### 1.3 性能基准测试

```cpp
// 性能测试框架
class Benchmark {
public:
    struct Stats {
        double avg_latency_ms;
        double p99_latency_ms;
        double qps;
    };

    static Stats benchmark_search(
        VectorSearch& index,
        const float* queries,
        size_t num_queries,
        size_t k,
        int num_threads = 1) {

        std::vector<double> latencies;
        latencies.reserve(num_queries);

        auto start_total = std::chrono::high_resolution_clock::now();

        for (size_t i = 0; i < num_queries; i++) {
            std::vector<float> distances(k);
            std::vector<size_t> labels(k);

            auto q_start = std::chrono::high_resolution_clock::now();
            index.search(&queries[i * index.dimension], k,
                        distances.data(), labels.data());
            auto q_end = std::chrono::high_resolution_clock::now();

            double latency = std::chrono::duration<
                double, std::milli>(q_end - q_start).count();
            latencies.push_back(latency);
        }

        auto end_total = std::chrono::high_resolution_clock::now();
        double total_time = std::chrono::duration<
            double>(end_total - start_total).count();

        Stats stats;
        stats.avg_latency_ms = std::accumulate(
            latencies.begin(), latencies.end(), 0.0) / num_queries;

        std::sort(latencies.begin(), latencies.end());
        stats.p99_latency_ms = latencies[size_t(num_queries * 0.99)];

        stats.qps = num_queries / total_time;

        return stats;
    }

    static void print_stats(const Stats& stats, const char* name) {
        printf("%s Performance:\n", name);
        printf("  Avg Latency: %.2f ms\n", stats.avg_latency_ms);
        printf("  P99 Latency: %.2f ms\n", stats.p99_latency_ms);
        printf("  QPS: %.2f\n", stats.qps);
        printf("\n");
    }
};

// 生成测试数据
std::vector<float> generate_random_vectors(size_t n, size_t d) {
    std::vector<float> vectors(n * d);
    std::random_device rd;
    std::mt19937 gen(rd());

    for (size_t i = 0; i < n * d; i++) {
        vectors[i] = dist(gen);
    }

    return vectors;
```

## 第二阶段：算法优化 (Day 2)

### 2.1 IVF 索引实现

```cpp
#include <vector>
#include <cmath>
#include <algorithm>

// IVF (Inverted File) 索引
class IVFVectorSearch : public VectorSearch {
    size_t dimension;
    size_t num_lists;      // 聚类中心数量
    size_t nprobe;         // 搜索时检查的聚类数

    // 聚类中心
    std::vector<float> centroids;  // [num_lists, d]

    // 倒排列表
    struct InvertedList {
        std::vector<float> vectors;  // [size, d]
        std::vector<size_t> ids;
    };
    std::vector<InvertedList> lists;

    // 量化器（简化版：使用真实距离）
    size_t find_nearest_centroid(const float* vector) const {
        float min_dist = std::numeric_limits<float>::max();
        size_t min_idx = 0;

        for (size_t i = 0; i < num_lists; i++) {
            float dist = 0;
            for (size_t j = 0; j < dimension; j++) {
                float diff = vector[j] - centroids[i * dimension + j];
                dist += diff * diff;
            }

            if (dist < min_dist) {
                min_dist = dist;
                min_idx = i;
            }
        }

        return min_idx;
    }

public:
    IVFVectorSearch(size_t d, size_t nlist = 100, size_t nprobe = 10)
        : dimension(d), num_lists(nlist), nprobe(nprobe) {

        // 初始化聚类中心（随机）
        centroids.resize(num_lists * dimension);
        for (size_t i = 0; i < num_lists * dimension; i++) {
            centroids[i] = ((float)rand() / RAND_MAX) * 2 - 1;
        }

        lists.resize(num_lists);
    }

    void add_vectors(const float* vectors, size_t count) override {
        // 分配到聚类
        for (size_t i = 0; i < count; i++) {
            size_t list_id = find_nearest_centroid(&vectors[i * dimension]);

            // 添加向量
            auto& list = lists[list_id];
            list.vectors.insert(list.vectors.end(),
                               &vectors[i * dimension],
                               &vectors[i * dimension] + dimension);
            list.ids.push_back(i);
        }
    }

    void search(
        const float* query,
        size_t k,
        float* distances,
        size_t* labels) const override {

        // 找到最近的聚类中心
        std::vector<std::pair<float, size_t>> centroid_distances;
        for (size_t i = 0; i < num_lists; i++) {
            float dist = 0;
            for (size_t j = 0; j < dimension; j++) {
                float diff = query[j] - centroids[i * dimension + j];
                dist += diff * diff;
            }
            centroid_distances.push_back({dist, i});
        }

        // 排序并选择前 nprobe 个
        std::sort(centroid_distances.begin(), centroid_distances.end());
        centroid_distances.resize(std::min(nprobe, num_lists));

        // 搜索选中的倒排表
        std::vector<std::pair<float, size_t>> all_distances;

        for (size_t idx = 0; idx < centroid_distances.size(); idx++) {
            size_t list_id = centroid_distances[idx].second;
            const auto& list = lists[list_id];

            // 扫描倒排表
            for (size_t i = 0; i < list.vectors.size() / dimension; i++) {
                float dist = 0;
                for (size_t j = 0; j < dimension; j++) {
                    float diff = query[j] - list.vectors[i * dimension + j];
                    dist += diff * diff;
                }
                all_distances.push_back({dist, list.ids[i]});
            }
        }

        // 排序并返回 Top-K
        std::sort(all_distances.begin(), all_distances.end());

        for (size_t i = 0; i < k && i < all_distances.size(); i++) {
            distances[i] = all_distances[i].first;
            labels[i] = all_distances[i].second;
        }
    }

    size_t size() const override {
        size_t total = 0;
        for (const auto& list : lists) {
            total += list.ids.size();
        }
        return total;
    }
};
```

## 第三阶段：数据布局优化 (Day 3)

### 3.1 转置存储

```cpp
// 优化：转置存储，提高缓存命中率
class IVFVectorSearchTransposed : public VectorSearch {
    size_t dimension;
    size_t num_lists;
    size_t nprobe;

    std::vector<float> centroids;  // [num_lists, d]
    std::vector<float> centroids_norms;  // [num_lists]

    struct InvertedList {
        // 转置存储：[d, size]
        std::vector<float> vectors;
        std::vector<size_t> ids;
        std::vector<float> norms;  // 预计算范数
        size_t size;
    };
    std::vector<InvertedList> lists;

public:
    IVFVectorSearchTransposed(size_t d, size_t nlist = 100, size_t nprobe = 10)
        : dimension(d), num_lists(nlist), nprobe(nprobe) {

        centroids.resize(num_lists * dimension);
        for (size_t i = 0; i < num_lists * dimension; i++) {
            centroids[i] = ((float)rand() / RAND_MAX) * 2 - 1;
        }

        // 预计算聚类中心范数
        centroids_norms.resize(num_lists);
        for (size_t i = 0; i < num_lists; i++) {
            float norm = 0;
            for (size_t j = 0; j < dimension; j++) {
                norm += centroids[i * dimension + j] * centroids[i * dimension + j];
            }
            centroids_norms[i] = norm;
        }

        lists.resize(num_lists);
    }

    void add_vectors(const float* vectors, size_t count) override {
        for (size_t i = 0; i < count; i++) {
            size_t list_id = find_nearest_centroid(&vectors[i * dimension]);
            auto& list = lists[list_id];

            // 转置存储
            for (size_t d = 0; d < dimension; d++) {
                list.vectors.push_back(vectors[i * dimension + d]);
            }

            // 预计算范数
            float norm = 0;
            for (size_t d = 0; d < dimension; d++) {
                float val = vectors[i * dimension + d];
                norm += val * val;
            }
            list.norms.push_back(norm);

            list.ids.push_back(i);
            list.size++;
        }
    }

    void search(
        const float* query,
        size_t k,
        float* distances,
        size_t* labels) const override {

        // 计算查询范数
        float query_norm = 0;
        for (size_t j = 0; j < dimension; j++) {
            query_norm += query[j] * query[j];
        }

        // 找最近的聚类中心
        std::vector<std::pair<float, size_t>> centroid_distances;
        for (size_t i = 0; i < num_lists; i++) {
            float dist = 0;
            for (size_t j = 0; j < dimension; j++) {
                float diff = query[j] - centroids[i * dimension + j];
                dist += diff * diff;
            }
            centroid_distances.push_back({dist, i});
        }

        std::sort(centroid_distances.begin(), centroid_distances.end());
        centroid_distances.resize(std::min(nprobe, num_lists));

        // 搜索倒排表（使用转置存储）
        std::vector<std::pair<float, size_t>> all_distances;

        for (size_t idx = 0; idx < centroid_distances.size(); idx++) {
            size_t list_id = centroid_distances[idx].second;
            const auto& list = lists[list_id];

            // 计算点积（转置存储，缓存友好）
            for (size_t i = 0; i < list.size; i++) {
                float dot = 0;
                for (size_t d = 0; d < dimension; d++) {
                    dot += query[d] * list.vectors[d * list.size + i];
                }

                // L2 距离：||q||² + ||v||² - 2*<q,v>
                float dist = query_norm + list.norms[i] - 2 * dot;
                all_distances.push_back({dist, list.ids[i]});
            }
        }

        std::sort(all_distances.begin(), all_distances.end());

        for (size_t i = 0; i < k && i < all_distances.size(); i++) {
            distances[i] = all_distances[i].first;
            labels[i] = all_distances[i].second;
        }
    }

    size_t size() const override {
        size_t total = 0;
        for (const auto& list : lists) {
            total += list.size;
        }
        return total;
    }

private:
    size_t find_nearest_centroid(const float* vector) const {
        float min_dist = std::numeric_limits<float>::max();
        size_t min_idx = 0;

        for (size_t i = 0; i < num_lists; i++) {
            float dist = 0;
            for (size_t j = 0; j < dimension; j++) {
                float diff = vector[j] - centroids[i * dimension + j];
                dist += diff * diff;
            }

            if (dist < min_dist) {
                min_dist = dist;
                min_idx = i;
            }
        }

        return min_idx;
    }
};
```

## 第四阶段：SIMD 优化 (Day 4)

### 4.1 SIMD 距离计算

```cpp
#include <immintrin.h>

// SIMD 优化的 IVF 索引
class IVFVectorSearchSIMD : public IVFVectorSearchTransposed {
public:
    IVFVectorSearchSIMD(size_t d, size_t nlist = 100, size_t nprobe = 10)
        : IVFVectorSearchTransposed(d, nlist, nprobe) {}

    void search(
        const float* query,
        size_t k,
        float* distances,
        size_t* labels) const override {

        // 计算查询范数
        float query_norm = 0;
        for (size_t j = 0; j < dimension; j++) {
            query_norm += query[j] * query[j];
        }

        // SIMD 找最近的聚类中心
        std::vector<std::pair<float, size_t>> centroid_distances;

        for (size_t base = 0; base < num_lists; base += 8) {
            size_t batch = std::min(8UL, num_lists - base);

            __m256 dists = _mm256_setzero_ps();
            __m256 qnorm = _mm256_set1_ps(query_norm);

            for (size_t d = 0; d < dimension; d++) {
                __m256 q_d = _mm256_set1_ps(query[d]);
                __m256 c_d = _mm256_loadu_ps(&centroids[d * num_lists + base]);

                __m256 diff = _mm256_sub_ps(q_d, c_d);
                dists = _mm256_fmadd_ps(diff, diff, dists);
            }

            __m256 cnorm = _mm256_loadu_ps(&centroids_norms[base]);

            // 完整距离：||q||² + ||c||² - 2*<q,c>
            // 这里简化为直接使用 dists
            // 实际应该用点积公式

            // 提取结果
            float dist_array[8];
            _mm256_storeu_ps(dist_array, dists);

            for (size_t i = 0; i < batch; i++) {
                centroid_distances.push_back({dist_array[i], base + i});
            }
        }

        std::sort(centroid_distances.begin(), centroid_distances.end());
        centroid_distances.resize(std::min(nprobe, num_lists));

        // 搜索倒排表（SIMD 批量点积）
        std::vector<std::pair<float, size_t>> all_distances;

        for (size_t idx = 0; idx < centroid_distances.size(); idx++) {
            size_t list_id = centroid_distances[idx].second;
            const auto& list = lists[list_id];

            // 批量计算点积（每次 8 个向量）
            for (size_t base = 0; base < list.size; base += 8) {
                size_t batch = std::min(8UL, list.size - base);

                __m256 dots = _mm256_setzero_ps();

                for (size_t d = 0; d < dimension; d++) {
                    __m256 q_d = _mm256_set1_ps(query[d]);
                    __m256 v_d = _mm256_loadu_ps(
                        &list.vectors[d * list.size + base]
                    );

                    dots = _mm256_fmadd_ps(q_d, v_d, dots);
                }

                // 提取并计算距离
                float dots_array[8];
                _mm256_storeu_ps(dots_array, dots);

                __m256 norms = _mm256_loadu_ps(&list.norms[base]);
                float norms_array[8];
                _mm256_storeu_ps(norms_array, norms);

                __m256 qnorm_vec = _mm256_set1_ps(query_norm);

                for (size_t i = 0; i < batch; i++) {
                    // L2 距离
                    float dist = query_norm + norms_array[i] - 2 * dots_array[i];
                    all_distances.push_back({dist, list.ids[base + i]});
                }
            }
        }

        std::sort(all_distances.begin(), all_distances.end());

        for (size_t i = 0; i < k && i < all_distances.size(); i++) {
            distances[i] = all_distances[i].first;
            labels[i] = all_distances[i].second;
        }
    }
};
```

### 5.1 OpenMP 并行搜索

```cpp
#include <omp.h>

// 多线程 IVF 索引
class IVFVectorSearchParallel : public IVFVectorSearchSIMD {
public:
    IVFVectorSearchParallel(size_t d, size_t nlist = 1000, size_t nprobe = 100)
        : IVFVectorSearchSIMD(d, nlist, nprobe) {}

    void search(
        const float* query,
        size_t k,
        float* distances,
        size_t* labels) const override {

        // 计算查询范数
        float query_norm = 0;
        for (size_t j = 0; j < dimension; j++) {
            query_norm += query[j] * query[j];
        }

        // 找最近的聚类中心（可以并行）
        std::vector<std::pair<float, size_t>> centroid_distances(num_lists);

        #pragma omp parallel for schedule(static)
        for (size_t i = 0; i < num_lists; i++) {
            float dist = 0;
            for (size_t j = 0; j < dimension; j++) {
                float diff = query[j] - centroids[i * dimension + j];
                dist += diff * diff;
            }
            centroid_distances[i] = {dist, i};
        }

        std::sort(centroid_distances.begin(), centroid_distances.end());
        centroid_distances.resize(std::min(nprobe, num_lists));

        // 并行搜索倒排表
        std::vector<std::vector<std::pair<float, size_t>>> results(
            centroid_distances.size()
        );

        #pragma omp parallel for schedule(dynamic)
        for (size_t idx = 0; idx < centroid_distances.size(); idx++) {
            size_t list_id = centroid_distances[idx].second;
            const auto& list = lists[list_id];

            for (size_t base = 0; base < list.size; base += 8) {
                size_t batch = std::min(8UL, list.size - base);

                __m256 dots = _mm256_setzero_ps();

                for (size_t d = 0; d < dimension; d++) {
                    __m256 q_d = _mm256_set1_ps(query[d]);
                    __m256 v_d = _mm256_loadu_ps(
                        &list.vectors[d * list.size + base]
                    );
                    dots = _mm256_fmadd_ps(q_d, v_d, dots);
                }

                float dots_array[8];
                _mm256_storeu_ps(dots_array, dots);

                __m256 norms = _mm256_loadu_ps(&list.norms[base]);
                float norms_array[8];
                _mm256_storeu_ps(norms_array, norms);

                for (size_t i = 0; i < batch; i++) {
                    float dist = query_norm + norms_array[i] - 2 * dots_array[i];
                    results[idx].push_back({dist, list.ids[base + i]});
                }
            }
        }

        // 合并结果
        std::vector<std::pair<float, size_t>> all_distances;
        for (auto& result : results) {
            all_distances.insert(
                all_distances.end(),
                result.begin(),
                result.end()
            );
        }

        std::sort(all_distances.begin(), all_distances.end());

        for (size_t i = 0; i < k && i < all_distances.size(); i++) {
            distances[i] = all_distances[i].first;
            labels[i] = all_distances[i].second;
        }
    }
};
```

```cpp
class aligned_allocator {
public:
    using value_type = float;

    aligned_allocator() noexcept {}

    template<typename U>
    aligned_allocator(const aligned_allocator<U>&) noexcept {}

    float* allocate(size_t n) {
        void* ptr = nullptr;
        if (posix_memalign(&ptr, 64, n * sizeof(float)) != 0) {
            throw std::bad_alloc();
        }
        return static_cast<float*>(ptr);
    }

    void deallocate(float* p, size_t) {
        free(p);
    }
};

// 使用内存池的 IVF 索引
class IVFVectorSearchOptimized : public IVFVectorSearchParallel {
    // 内存池
    struct MemoryPool {
        std::stack<float*> free_blocks;
        size_t block_size;

        MemoryPool(size_t bs) : block_size(bs) {}

        float* allocate() {
            if (free_blocks.empty()) {
                return aligned_allocator().allocate(block_size);
            }
            float* ptr = free_blocks.top();
            free_blocks.pop();
            return ptr;
        }

        void deallocate(float* ptr) {
            free_blocks.push(ptr);
        }
    };

    MemoryPool pool;

public:
    IVFVectorSearchOptimized(size_t d, size_t nlist = 1000, size_t nprobe = 100)
        : IVFVectorSearchParallel(d, nlist, nprobe),
          pool(d * 1000) {}  // 预分配 1000 个向量的块
```

### 7.1 完整的优化实现

```cpp
// 最终优化版本：整合所有技术
class UltraFastVectorSearch : public VectorSearch {
    size_t dimension;
    size_t num_lists;
    size_t nprobe;

    // 数据结构优化
    struct alignas(64) InvertedList {
        // 转置存储，64 字节对齐
        std::vector<float, aligned_allocator<float>> vectors;  // [d, size]
        std::vector<float, aligned_allocator<float>> norms;     // [size]
        std::vector<size_t> ids;                                 // [size]
        size_t size;

        InvertedList(size_t d, size_t capacity = 0) {
            vectors.reserve(d * capacity);
            norms.reserve(capacity);
            ids.reserve(capacity);
        }
    };

    std::vector<InvertedList> lists;
    std::vector<float, aligned_allocator<float>> centroids;
    std::vector<float, aligned_allocator<float>> centroids_norms;

    // 线程本地缓存
    mutable thread_local std::vector<float> query_buffer;
    mutable thread_local std::vector<float> distance_buffer;

public:
    UltraFastVectorSearch(size_t d, size_t nlist = 1000, size_t nprobe = 100)
        : dimension(d), num_lists(nlist), nprobe(nprobe) {

        // 分配聚类中心
        centroids.resize(num_lists * d);
        centroids_norms.resize(num_lists);

        // K-means 初始化（简化：随机）
        #pragma omp parallel for
        for (size_t i = 0; i < num_lists; i++) {
            for (size_t j = 0; j < d; j++) {
                centroids[i * d + j] = ((float)rand() / RAND_MAX) * 2 - 1;
            }

            // 预计算范数
            float norm = 0;
            for (size_t j = 0; j < d; j++) {
                float val = centroids[i * d + j];
                norm += val * val;
            }
            centroids_norms[i] = norm;
        }

        lists.resize(nlist);
    }

    void add_vectors(const float* vectors, size_t count) override {
        // 预分配空间
        size_t avg_per_list = count / num_lists;
        for (auto& list : lists) {
            list = InvertedList(dimension, avg_per_list);
        }

        // 并行添加向量
        #pragma omp parallel for schedule(dynamic)
        for (size_t i = 0; i < count; i++) {
            // 找最近的聚类中心
            float min_dist = std::numeric_limits<float>::max();
            size_t min_idx = 0;

            // 部分搜索（只检查部分聚类）
            for (size_t c = 0; c < num_lists; c += 10) {
                float dist = 0;
                for (size_t j = 0; j < dimension; j++) {
                    float diff = vectors[i * dimension + j] - centroids[c * dimension + j];
                    dist += diff * diff;
                }

                #pragma omp critical
                {
                    if (dist < min_dist) {
                        min_dist = dist;
                        min_idx = c;
                    }
                }
            }

            // 添加到倒排表
            #pragma omp critical
            {
                auto& list = lists[min_idx];
                for (size_t j = 0; j < dimension; j++) {
                    list.vectors.push_back(vectors[i * dimension + j]);
                }

                float norm = 0;
                for (size_t j = 0; j < dimension; j++) {
                    float val = vectors[i * dimension + j];
                    norm += val * val;
                }
                list.norms.push_back(norm);
                list.ids.push_back(i);
                list.size++;
            }
        }
    }

    void search(
        const float* query,
        size_t k,
        float* distances,
        size_t* labels) const override {

        // 计算查询范数
        float query_norm = 0;
        for (size_t j = 0; j < dimension; j++) {
            query_norm += query[j] * query[j];
        }

        // SIMD 找最近的聚类中心
        std::vector<std::pair<float, size_t>> centroid_distances;
        centroid_distances.reserve(num_lists);

        for (size_t base = 0; base < num_lists; base += 8) {
            size_t batch = std::min(8UL, num_lists - base);

            __m256 qnorm = _mm256_set1_ps(query_norm);
            __m256 cnorms = _mm256_loadu_ps(&centroids_norms[base]);
            __m256 min_dists = _mm256_set1_ps(1e30f);

            for (size_t d = 0; d < dimension; d++) {
                __m256 q_d = _mm256_set1_ps(query[d]);
                __m256 c_d = _mm256_loadu_ps(&centroids[d * num_lists + base]);

                __m256 diff = _mm256_sub_ps(q_d, c_d);
                __m256 sq = _mm256_mul_ps(diff, diff);

                // L2 距离：||q||² + ||c||² - 2*<q,c>
                // 简化：用距离平方
                min_dists = _mm256_add_ps(min_dists, sq);
            }

            // 添加 cnorms
            min_dists = _mm256_add_ps(min_dists, cnorms);

            // 提取结果
            float dists[8];
            _mm256_storeu_ps(dists, min_dists);

            for (size_t i = 0; i < batch; i++) {
                centroid_distances.push_back({dists[i], base + i});
            }
        }

        std::sort(centroid_distances.begin(), centroid_distances.end());
        centroid_distances.resize(std::min(nprobe, num_lists));

        // 并行搜索倒排表
        size_t total_candidates = 0;
        for (size_t idx = 0; idx < centroid_distances.size(); idx++) {
            total_candidates += lists[centroid_distances[idx].second].size;
        }

        std::vector<std::pair<float, size_t>> all_distances;
        all_distances.reserve(total_candidates);

        #pragma omp parallel for schedule(dynamic)
        for (size_t idx = 0; idx < centroid_distances.size(); idx++) {
            size_t list_id = centroid_distances[idx].second;
            const auto& list = lists[list_id];

            std::vector<std::pair<float, size_t>> local_results;
            local_results.reserve(list.size);

            // 批量处理
            for (size_t base = 0; base < list.size; base += 8) {
                size_t batch = std::min(8UL, list.size - base);

                __m256 dots = _mm256_setzero_ps();
                __m256 cnorms = _mm256_loadu_ps(&list.norms[base]);
                __m256 qnorm = _mm256_set1_ps(query_norm);

                for (size_t d = 0; d < dimension; d++) {
                    __m256 q_d = _mm256_set1_ps(query[d]);
                    __m256 v_d = _mm256_loadu_ps(&list.vectors[d * list.size + base]);

                    dots = _mm256_fmadd_ps(q_d, v_d, dots);
                }

                // 计算距离
                __m256 two_dots = _mm256_mul_ps(dots, _mm256_set1_ps(2.0f));
                __m256 dists = _mm256_add_ps(qnorm, cnorms);
                dists = _mm256_sub_ps(dists, two_dots);

                // 提取
                float dist_array[8];
                _mm256_storeu_ps(dist_array, dists);

                for (size_t i = 0; i < batch; i++) {
                    local_results.push_back({dist_array[i], list.ids[base + i]});
                }
            }

            // 合并结果
            #pragma omp critical
            {
                all_distances.insert(
                    all_distances.end(),
                    local_results.begin(),
                    local_results.end()
                );
            }
        }

        // 最终排序
        std::sort(all_distances.begin(), all_distances.end());

        // 返回 Top-K
        size_t result_size = std::min(k, all_distances.size());
        for (size_t i = 0; i < result_size; i++) {
            distances[i] = all_distances[i].first;
            labels[i] = all_distances[i].second;
        }
    }

    size_t size() const override {
        size_t total = 0;
        for (const auto& list : lists) {
            total += list.size;
        }
        return total;
    }
};
```

### 10.1 IndexIVFFlat完整实现

```cpp
#include <faiss/IndexIVFFlat.h>
#include <faiss/IndexFlat.h>
#include <faiss/invlists/InvertedLists.h>

namespace faiss {

// IndexIVFFlat: 带有存储向量的倒排文件索引
// 继承自IndexIVF，提供最精确的IVF搜索
struct IndexIVFFlat : IndexIVF {
    // 构造函数
    // @param quantizer: 粗量化器，通常是IndexFlat
    // @param d: 向量维度
    // @param nlist_: 倒排表数量
    // @param metric: 距离度量类型
    IndexIVFFlat(
            Index* quantizer,
            size_t d,
            size_t nlist_,
            MetricType metric = METRIC_L2,
            bool own_invlists = true)
        : IndexIVF(quantizer, d, nlist_, d * sizeof(float), metric, own_invlists) {
        // code_size = d * sizeof(float) 因为存储原始float向量
        is_trained = false;
    }

    // 核心添加向量实现
    void add_core(
            idx_t n,
            const float* x,
            const idx_t* xids,
            const idx_t* precomputed_idx,
            void* inverted_list_context) override {

        // 1. 计算或使用预计算的聚类分配
        std::vector<idx_t> idx(n);
        const idx_t* idx_ptr = precomputed_idx;

        if (!idx_ptr) {
            // 使用量化器计算聚类分配
            quantizer->assign(n, x, idx.data());
            idx_ptr = idx.data();
        }

        // 2. 编码向量（对于IVFFlat，直接存储原始float）
        std::vector<uint8_t> codes(n * code_size);
        encode_vectors(n, x, idx_ptr, codes.data());

        // 3. 添加到倒排表
        invlists->add_entries(
            n,                      // 添加n个向量
            idx_ptr,                // 每个向量对应的倒排表ID
            xids,                   // 向量的外部ID
            codes.data()            // 编码后的向量数据
        );

        ntotal += n;
    }

    // 编码向量：对于IVFFlat，直接复制float数据
    void encode_vectors(
            idx_t n,
            const float* x,
            const idx_t* list_nos,
            uint8_t* codes,
            bool include_listno) const override {

        // IVFFlat直接存储原始float向量，不进行量化
        // 只需要复制数据
        for (idx_t i = 0; i < n; i++) {
            if (list_nos[i] < 0) {
                continue; // 跳过无效的分配
            }

            // 复制向量数据
            const float* src = x + i * d;
            float* dst = (float*)(codes + i * code_size);

            if (by_residual && metric_type == METRIC_L2) {
                // 如果使用残差编码，先减去聚类中心
                idx_t list_no = list_nos[i];
                float* centroid = quantizer->get_xb() + list_no * d;

                for (size_t j = 0; j < d; j++) {
                    dst[j] = src[j] - centroid[j];
                }
            } else {
                // 直接复制原始向量
                memcpy(dst, src, d * sizeof(float));
            }
        }
    }

    // 获取倒排表扫描器
    InvertedListScanner* get_InvertedListScanner(
            bool store_pairs,
            const IDSelector* sel,
            const IVFSearchParameters* params) const override {

        // 根据度量类型创建不同的扫描器
        if (metric_type == METRIC_L2) {
            return get_L2sq_scanner(store_pairs, sel);
        } else if (metric_type == METRIC_INNER_PRODUCT) {
            return get_IP_scanner(store_pairs, sel);
        } else {
            return get_generic_scanner(store_pairs, sel);
        }
    }

    // L2距离扫描器实现
    struct IVFFlatL2Scanner : InvertedListScanner {
        const IndexIVFFlat& ivf;
        const float* query;         // 查询向量
        float query_norm;           // 查询向量的L2范数平方

        IVFFlatL2Scanner(
            const IndexIVFFlat& ivf,
            bool store_pairs = false,
            const IDSelector* sel = nullptr)
            : InvertedListScanner(store_pairs, sel), ivf(ivf) {}

        void set_query(const float* x) override {
            query = x;
            // 预计算查询向量范数
            query_norm = 0;
            for (size_t i = 0; i < ivf.d; i++) {
                query_norm += x[i] * x[i];
            }
        }

        void set_list(idx_t list_no, float coarse_dis) override {
            this->list_no = list_no;
            // 可以在这里预加载聚类中心
        }

        // 计算单个code的距离
        float distance_to_code(const uint8_t* code) const override {
            const float* vec = (const float*)code;

            if (ivf.by_residual) {
                // 如果存储的是残差，需要加上聚类中心
                const float* centroid = ivf.quantizer->get_xb() + list_no * ivf.d;

                float dis = query_norm;
                for (size_t i = 0; i < ivf.d; i++) {
                    float residual = vec[i];
                    float full_vec = residual + centroid[i];
                    dis += full_vec * full_vec - 2 * query[i] * full_vec;
                }
                return dis;
            } else {
                // 直接计算L2距离
                float dis = query_norm;
                for (size_t i = 0; i < ivf.d; i++) {
                    float diff = query[i] - vec[i];
                    dis += diff * diff;
                }
                return dis;
            }
        }

        // 批量扫描codes（优化版本）
        size_t scan_codes(
                size_t n,
                const uint8_t* codes,
                const idx_t* ids,
                float* distances,
                idx_t* labels,
                size_t k) const override {

            size_t nup = 0;
            const float* vecs = (const float*)codes;

            if (ivf.by_residual) {
                const float* centroid = ivf.quantizer->get_xb() + list_no * ivf.d;

                // 优化：展开循环
                for (size_t i = 0; i < n; i++) {
                    if (sel && !sel->is_member(ids[i])) {
                        continue;
                    }

                    const float* vec = vecs + i * ivf.d;
                    float dis = query_norm;

                    // SIMD优化的距离计算
                    for (size_t j = 0; j < ivf.d; j++) {
                        float residual = vec[j];
                        float full_vec = residual + centroid[j];
                        dis += full_vec * full_vec - 2 * query[j] * full_vec;
                    }

                    // 更新堆
                    if (dis < distances[0]) {
                        maxheap_replace_top<CMax<float, idx_t>>(
                            k, distances, labels, dis,
                            store_pairs ? (list_no << 32 | i) : ids[i]
                        );
                        nup++;
                    }
                }
            } else {
                // 不使用残差的版本
                for (size_t i = 0; i < n; i++) {
                    if (sel && !sel->is_member(ids[i])) {
                        continue;
                    }

                    const float* vec = vecs + i * ivf.d;
                    float dis = query_norm;

                    for (size_t j = 0; j < ivf.d; j++) {
                        float diff = query[j] - vec[j];
                        dis += diff * diff;
                    }

                    if (dis < distances[0]) {
                        maxheap_replace_top<CMax<float, idx_t>>(
                            k, distances, labels, dis,
                            store_pairs ? (list_no << 32 | i) : ids[i]
                        );
                        nup++;
                    }
                }
            }

            return nup;
        }
    };

    InvertedListScanner* get_L2sq_scanner(
            bool store_pairs,
            const IDSelector* sel) const {
        return new IVFFlatL2Scanner(*this, store_pairs, sel);
    }
};
```

```cpp
        codes[list_no].resize(new_size * code_size);

        memcpy(ids[list_no].data() + o, xids, n_entry * sizeof(idx_t));
        memcpy(codes[list_no].data() + o * code_size, code, n_entry * code_size);

        return o;

            size_t offset,
            size_t n_entry,

            resize(list_no, offset + n_entry);

        memcpy(ids[list_no].data() + offset, xids, n_entry * sizeof(idx_t));
        memcpy(codes[list_no].data() + offset * code_size, code, n_entry * code_size);
    }

        }
```

```cpp
            printf("Clustering objective: %g\n", clus.obj);

    void encode_listno(idx_t list_no, uint8_t* code) const {
        // 根据nlist大小决定编码方式
        if (nlist <= 256) {
            code[0] = list_no & 0xff;
        } else if (nlist <= 65536) {
            code[0] = list_no & 0xff;
            code[1] = (list_no >> 8) & 0xff;
        } else {
            code[0] = list_no & 0xff;
            code[1] = (list_no >> 8) & 0xff;
            code[2] = (list_no >> 16) & 0xff;
            code[3] = (list_no >> 24) & 0xff;
        }
    }

    // 解码列表号
    idx_t decode_listno(const uint8_t* code) const {
        if (nlist <= 256) {
            return code[0];
        } else if (nlist <= 65536) {
            return code[0] | (code[1] << 8);
        } else {
            return code[0] | (code[1] << 8) | (code[2] << 16) | (code[3] << 24);
        }
    }

    // 计算coarse code大小
    size_t coarse_code_size() const {
        if (nlist <= 256) {
            return 1;
        } else if (nlist <= 65536) {
            return 2;
        } else {
            return 4;
        }
    }
```

```cpp
namespace faiss {

// IndexIVF的search_preassigned实现
void IndexIVF::search_preassigned(
        idx_t n,
        const float* x,
        idx_t k,
        const idx_t* assign,
        const float* centroid_dis,
        float* distances,
        idx_t* labels,
        bool store_pairs,
        const IVFSearchParameters* params,
        IndexIVFStats* stats) const {

    // 1. 初始化搜索参数
    size_t nprobe = params ? params->nprobe : this->nprobe;
    size_t max_codes = params ? params->max_codes : this->max_codes;

    // 2. 初始化堆（每个查询一个大小为k的堆）
    float_minheap_array_t res;
    res.nh = n;
    res.k = k;
    res.ids = labels;
    res.val = distances;
    res.heapify();

    // 3. 并行搜索倒排表
    if (parallel_mode == 0) {
        // 模式0：按查询并行
        #pragma omp parallel for if (n > 100)
        for (idx_t i = 0; i < n; i++) {
            // 获取扫描器
            std::unique_ptr<InvertedListScanner> scanner(
                get_InvertedListScanner(store_pairs, params ? params->sel : nullptr)
            );

            // 设置查询
            scanner->set_query(x + i * d);

            // 扫描nprobe个倒排表
            size_t nscan = 0;
            for (size_t m = 0; m < nprobe; m++) {
                idx_t list_no = assign[i * nprobe + m];

                if (list_no < 0 || list_no >= nlist) {
                    continue;
                }

                size_t list_size = invlists->list_size(list_no);
                if (list_size == 0) {
                    continue;
                }

                // 获取倒排表数据
                InvertedLists::ScopedCodes codes(invlists, list_no);
                InvertedLists::ScopedIds ids(invlists, list_no);

                // 设置当前列表
                scanner->set_list(list_no, centroid_dis[i * nprobe + m]);

                // 扫描codes
                scanner->scan_codes(
                    list_size,
                    codes.get(),
                    ids.get(),
                    res.get_val(i),
                    res.get_ids(i),
                    k
                );

                nscan += list_size;
                if (max_codes > 0 && nscan >= max_codes) {
                    break;
                }
            }
        }
    } else {
        // 模式1：按倒排表并行
        for (idx_t i = 0; i < n; i++) {
            std::unique_ptr<InvertedListScanner> scanner(
                get_InvertedListScanner(store_pairs, params ? params->sel : nullptr)
            );
            scanner->set_query(x + i * d);

            #pragma omp parallel for
            for (size_t m = 0; m < nprobe; m++) {
                idx_t list_no = assign[i * nprobe + m];

                if (list_no < 0 || list_no >= nlist) {
                    continue;
                }

                size_t list_size = invlists->list_size(list_no);
                if (list_size == 0) {
                    continue;
                }

                InvertedLists::ScopedCodes codes(invlists, list_no);
                InvertedLists::ScopedIds ids(invlists, list_no);

                scanner->set_list(list_no, centroid_dis[i * nprobe + m]);

                #pragma omp critical
                {
                    scanner->scan_codes(
                        list_size,
                        codes.get(),
                        ids.get(),
                        res.get_val(i),
                        res.get_ids(i),
                        k
                    );
                }
            }
        }
    }

    // 4. 整理堆结果
    res.reorder();
}
```

### 10.5 Heap操作优化

```cpp
#include <faiss/utils/Heap.h>

namespace faiss {

// 使用Faiss的堆操作实现Top-K搜索
template <typename C>
inline void heap_search(
        const float* query,
        size_t d,
        const float* vectors,
        size_t n,
        size_t k,
        float* distances,
        idx_t* labels) {

    // 1. 初始化堆
    float* bh_val = distances;
    idx_t* bh_ids = labels;

    // 堆初始化：前k个元素
    heap_heapify<C>(k, bh_val, bh_ids);

    // 添加前k个元素
    for (size_t i = 0; i < k && i < n; i++) {
        float dis = 0;
        for (size_t j = 0; j < d; j++) {
            float diff = query[j] - vectors[i * d + j];
            dis += diff * diff;
        }
        heap_push<C>(i + 1, bh_val, bh_ids, dis, i);
    }

    // 2. 处理剩余元素
    for (size_t i = k; i < n; i++) {
        float dis = 0;
        for (size_t j = 0; j < d; j++) {
            float diff = query[j] - vectors[i * d + j];
            dis += diff * diff;
        }

        // 如果距离比堆顶小，替换堆顶
        if (C::cmp(bh_val[0], dis)) {
            heap_replace_top<C>(k, bh_val, bh_ids, dis, i);
        }
    }

    // 3. 整理堆（将堆转换为有序数组）
    heap_reorder<C>(k, bh_val, bh_ids);
}

// SIMD优化的距离计算+堆操作
template <typename C>
inline void heap_search_simd(
        const float* query,
        size_t d,
        const float* vectors,
        size_t n,
        size_t k,
        float* distances,
        idx_t* labels) {

    float* bh_val = distances;
    idx_t* bh_ids = labels;

    heap_heapify<C>(k, bh_val, bh_ids);

    // 批量处理（每次8个向量）
    for (size_t i = 0; i < n; i += 8) {
        size_t batch = std::min(8UL, n - i);

        // SIMD批量计算距离
        __m256 diss[4]; // 假设d是32的倍数
        for (size_t j = 0; j < d; j += 32) {
            __m256 q0 = _mm256_set1_ps(query[j]);
            __m256 q1 = _mm256_set1_ps(query[j + 8]);
            __m256 q2 = _mm256_set1_ps(query[j + 16]);
            __m256 q3 = _mm256_set1_ps(query[j + 24]);

            for (size_t b = 0; b < batch; b++) {
                __m256 v0 = _mm256_loadu_ps(vectors + (i + b) * d + j);
                __m256 v1 = _mm256_loadu_ps(vectors + (i + b) * d + j + 8);
                __m256 v2 = _mm256_loadu_ps(vectors + (i + b) * d + j + 16);
                __m256 v3 = _mm256_loadu_ps(vectors + (i + b) * d + j + 24);

                __m256 diff0 = _mm256_sub_ps(q0, v0);
                __m256 diff1 = _mm256_sub_ps(q1, v1);
                __m256 diff2 = _mm256_sub_ps(q2, v2);
                __m256 diff3 = _mm256_sub_ps(q3, v3);

                __m256 sq0 = _mm256_mul_ps(diff0, diff0);
                __m256 sq1 = _mm256_mul_ps(diff1, diff1);
                __m256 sq2 = _mm256_mul_ps(diff2, diff2);
                __m256 sq3 = _mm256_mul_ps(diff3, diff3);

                if (j == 0) {
                    diss[b] = _mm256_add_ps(sq0, sq1);
                    diss[b] = _mm256_add_ps(diss[b], sq2);
                    diss[b] = _mm256_add_ps(diss[b], sq3);
                } else {
                    diss[b] = _mm256_add_ps(diss[b], sq0);
                    diss[b] = _mm256_add_ps(diss[b], sq1);
                    diss[b] = _mm256_add_ps(diss[b], sq2);
                    diss[b] = _mm256_add_ps(diss[b], sq3);
                }
            }
        }

        // 水平求和
        float dists[8];
        for (size_t b = 0; b < batch; b++) {
            dists[b] = _mm256_reduce_add_ps(diss[b]);
        }

        // 更新堆
        for (size_t b = 0; b < batch; b++) {
            if (i + b < k) {
                heap_push<C>(i + b + 1, bh_val, bh_ids, dists[b], i + b);
            } else if (C::cmp(bh_val[0], dists[b])) {
                heap_replace_top<C>(k, bh_val, bh_ids, dists[b], i + b);
            }
        }
    }

    heap_reorder<C>(k, bh_val, bh_ids);
}
```

```cpp
   - max_codes: 限制最大访问code数量
   - parallel_mode: 选择合适的并行策略

3. 数据布局优化
   - by_residual: 使用残差编码减少量化误差
   - 转置存储：提高缓存命中率

4. 并行化策略
   - 模式0：按查询并行（适合少量查询）
   - 模式1：按倒排表并行（适合大量查询）
   - 模式2：混合并行

5. 内存优化
   - 预分配内存避免频繁分配
   - 使用64字节对齐提高SIMD性能
   - 考虑使用GPU加速

6. 索引选择
   - IndexFlat: 精确搜索，小规模数据
   - IndexIVFFlat: 平衡精度和速度
   - IndexIVFPQ: 大规模数据，内存受限
   - IndexHNSW: 高精度要求
*/

// 参数选择指南
struct IVFParameterGuide {
    static size_t suggest_nlist(size_t num_vectors) {
        // 经验公式：nlist = sqrt(N)
        return std::max(1UL, (size_t)std::sqrt(num_vectors));
    }

    static size_t suggest_nprobe(size_t nlist, double target_recall) {
        // nprobe影响召回率
        // recall=0.9: nprobe ≈ nlist / 10
        // recall=0.95: nprobe ≈ nlist / 5
        // recall=0.99: nprobe ≈ nlist / 2

        if (target_recall >= 0.99) {
            return nlist / 2;
        } else if (target_recall >= 0.95) {
            return nlist / 5;
        } else {
            return nlist / 10;
        }
    }

    static int suggest_threads(size_t num_queries) {
        // 少量查询：线程数=查询数
        // 大量查询：线程数=CPU核心数
        int num_cores = omp_get_max_threads();
        return std::min(num_cores, (int)num_queries);
    }
};
```

```cpp
// 生产环境封装
class FaissVectorIndex {
    faiss::Index* index = nullptr;
    std::mutex mutex;  // 线程安全

public:
    FaissVectorIndex(const std::string& index_desc) {
        // 使用index_factory创建索引
        // 例如："IVF1024,Flat" 表示1024个聚类中心的IVFFlat
        index = faiss::index_factory(128, index_desc.c_str());
    }

    void train_and_add(const float* vectors, size_t n) {
        std::lock_guard<std::mutex> lock(mutex);

        // 分批训练
        size_t train_size = std::min(n, 1000000UL);
        index->train(train_size, vectors);

        // 分批添加
        size_t batch_size = 100000;
        for (size_t i = 0; i < n; i += batch_size) {
            size_t batch = std::min(batch_size, n - i);
            index->add(batch, vectors + i * 128);
        }
    }

    void search(
            const float* query,
            size_t k,
            float* distances,
            idx_t* labels) {

        // Faiss的search是线程安全的，不需要加锁
        index->search(1, query, k, distances, labels);
    }

    // 保存索引
    void save(const std::string& filename) {
        std::lock_guard<std::mutex> lock(mutex);
        faiss::write_index(index, filename.c_str());
    }

    // 加载索引
    void load(const std::string& filename) {
        std::lock_guard<std::mutex> lock(mutex);
        if (index) {
            delete index;
        }
        index = faiss::read_index(filename.c_str());
    }

    ~FaissVectorIndex() {
        delete index;
    }
};
```
